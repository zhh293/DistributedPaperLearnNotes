# 订单系统架构设计

## 一、问题背景

### 1.1 业务场景

在大型 O2O 平台中，订单系统是整个交易链路的核心枢纽。一个典型的外卖/到店订单，需要经历用户下单、商家接单、备餐、配送、完成/退款等多个阶段。随着业务规模的增长，订单系统面临以下核心挑战：

1. **状态爆炸问题**：早期系统将商家状态、物流状态、退款状态聚合到单一状态码中，导致状态值数量呈笛卡尔积增长，维护成本极高。
2. **多角色视角差异**：用户、商家、骑手、客服对同一订单的关注点不同，需要不同的状态展示逻辑。
3. **高并发处理**：大促期间峰值 QPS 可达数十万，订单创建、状态流转必须保证高可用。
4. **数据一致性**：订单涉及支付、库存、优惠券等多个子系统，需要保证分布式环境下的最终一致性。
5. **履约流程复杂**：接单、备餐、配送涉及多方协作，任何环节异常都需要有完善的补偿机制。

### 1.2 旧系统痛点分析

旧的订单系统采用单一状态码设计，一个 `orderStatus` 字段承载了所有维度的状态信息：

```java
/**
 * 旧系统状态码设计（反面教材）
 * 将商家、物流、退款状态混合在单一字段中
 */
public class LegacyOrderStatus {
    // 商家相关
    public static final int WAIT_ACCEPT = 1;       // 待接单
    public static final int ACCEPTED = 2;           // 已接单
    public static final int REJECTED = 3;           // 拒绝接单
    public static final int PREPARING = 4;          // 备餐中
    public static final int PREPARED = 5;           // 已出餐

    // 物流相关
    public static final int WAIT_RIDER = 10;        // 待分配骑手
    public static final int RIDER_ACCEPTED = 11;    // 骑手已接单
    public static final int RIDER_ARRIVED = 12;     // 骑手已到店
    public static final int RIDER_PICKED = 13;      // 骑手已取餐
    public static final int RIDER_DELIVERED = 14;    // 骑手已送达

    // 退款相关 - 与上面组合后状态爆炸
    public static final int REFUND_APPLYING = 20;   // 退款申请中
    public static final int REFUND_ACCEPTED_WAIT_RIDER = 21;  // 已接单+退款中
    public static final int REFUND_PREPARING_WAIT_RIDER = 22; // 备餐中+退款中
    // ... 更多组合状态，最终达到 50+ 种状态码
}
```

这种设计的核心问题在于：

1. **可维护性差**：每增加一个新维度的状态，都需要与已有状态做组合。
2. **扩展困难**：新业务线接入时，状态码体系需要重新设计。
3. **查询复杂**：根据特定维度的状态筛选订单变得非常困难。
4. **推送重复**：状态变更时，由于缺乏维度隔离，容易触发重复的消息推送。

### 1.3 设计目标

新的订单系统需要达到以下设计目标：

| 目标维度 | 具体要求 |
|---------|---------|
| 状态管理 | 多维度状态独立管理，避免状态爆炸 |
| 高可用 | 99.99% 可用性，支持峰值 50w QPS |
| 一致性 | 分布式环境下保证最终一致性 |
| 可扩展 | 支持新业务线快速接入 |
| 可观测 | 全链路追踪，状态变更可审计 |

#### 3.1.2 状态机引擎实现

```java
/**
 * 通用状态机引擎
 * 支持多维度状态独立流转，以及跨维度状态联动
 */
public class OrderStateMachine {

    /**
     * 状态转换规则定义
     */
    private final Map<StatusTransitionKey, StatusTransitionAction> transitionRules = new ConcurrentHashMap<>();

    /**
     * 状态变更监听器
     */
    private final List<StatusChangeListener> listeners = new CopyOnWriteArrayList<>();

    /**
     * 注册状态转换规则
     */
    public void registerTransition(StatusDimension dimension,
                                   int fromStatus,
                                   int toStatus,
                                   StatusTransitionAction action) {
        StatusTransitionKey key = new StatusTransitionKey(dimension, fromStatus, toStatus);
        transitionRules.put(key, action);
    }

    /**
     * 执行状态转换
     * @param context 订单上下文
     * @param dimension 状态维度
     * @param targetStatus 目标状态
     * @return 转换结果
     */
    @Transactional(rollbackFor = Exception.class)
    public TransitionResult transition(OrderContext context,
                                       StatusDimension dimension,
                                       int targetStatus) {
        int currentStatus = context.getCurrentStatus(dimension);

        // 1. 查找转换规则
        StatusTransitionKey key = new StatusTransitionKey(dimension, currentStatus, targetStatus);
        StatusTransitionAction action = transitionRules.get(key);
        if (action == null) {
            return TransitionResult.fail("非法状态转换: " + dimension
                    + " from " + currentStatus + " to " + targetStatus);
        }

        // 2. 前置校验
        if (!action.preCheck(context)) {
            return TransitionResult.fail("前置校验失败");
        }

        // 3. 记录上一次状态（用于防重复推送）
        long lastOrderStatus = context.getOrder().getLastOrderStatus();

        // 4. 执行状态变更
        action.execute(context);
        context.setStatus(dimension, targetStatus);

        // 5. 持久化
        orderRepository.updateStatus(context.getOrderId(), dimension, targetStatus);

        // 6. 发布状态变更事件
        StatusChangeEvent event = new StatusChangeEvent(
                context.getOrderId(),
                dimension,
                currentStatus,
                targetStatus,
                lastOrderStatus
        );
        notifyListeners(event);

        return TransitionResult.success();
    }

    private void notifyListeners(StatusChangeEvent event) {
        for (StatusChangeListener listener : listeners) {
            try {
                listener.onStatusChanged(event);
            } catch (Exception e) {
                log.error("状态变更监听器执行异常, orderId={}, listener={}",
                        event.getOrderId(), listener.getClass().getSimpleName(), e);
            }
        }
    }
}

/**
```
```java
 * 商家状态转换规则配置

@Configuration
public class MerchantStatusTransitionConfig {

    @Autowired
    private OrderStateMachine stateMachine;

    @PostConstruct
    public void init() {
        // 待接单 -> 已接单
        stateMachine.registerTransition(
                StatusDimension.MERCHANT,
                MerchantStatusEnum.WAIT_ACCEPT.getCode(),
                MerchantStatusEnum.ACCEPTED.getCode(),
                new AcceptOrderAction()
        );

        // 待接单 -> 拒绝接单
        stateMachine.registerTransition(
                StatusDimension.MERCHANT,
                MerchantStatusEnum.WAIT_ACCEPT.getCode(),
                MerchantStatusEnum.REJECTED.getCode(),
                new RejectOrderAction()
        );

        // 已接单 -> 待出餐
        stateMachine.registerTransition(
                StatusDimension.MERCHANT,
                MerchantStatusEnum.ACCEPTED.getCode(),
                MerchantStatusEnum.WAIT_PREPARE.getCode(),
                new StartPreparationAction()
        );

        // 待出餐 -> 已出餐
        stateMachine.registerTransition(
                StatusDimension.MERCHANT,
                MerchantStatusEnum.WAIT_PREPARE.getCode(),
                MerchantStatusEnum.PREPARED.getCode(),
                new MealPreparedAction()
        );

        // 待出餐 -> 出餐超时
        stateMachine.registerTransition(
                StatusDimension.MERCHANT,
                MerchantStatusEnum.WAIT_PREPARE.getCode(),
                MerchantStatusEnum.PREPARE_TIMEOUT.getCode(),
                new PreparationTimeoutAction()
        );

        // 已出餐 -> 履约完成
        stateMachine.registerTransition(
                StatusDimension.MERCHANT,
                MerchantStatusEnum.PREPARED.getCode(),
                MerchantStatusEnum.FULFILL_DONE.getCode(),
                new FulfillmentCompleteAction()
        );
    }
}

/**
 * 接单动作实现
 */
public class AcceptOrderAction implements StatusTransitionAction {

    @Override
    public boolean preCheck(OrderContext context) {
        // 检查商家是否在营业时间
        if (!context.getMerchant().isOpen()) {
            throw new BusinessException("商家不在营业时间");
        }
        // 检查订单是否已超时未接单
        long waitMinutes = Duration.between(
                context.getOrder().getCreateTime().toInstant(),
                Instant.now()
        ).toMinutes();
        if (waitMinutes > 15) {
            throw new BusinessException("订单已超时，无法接单");
        }
        return true;
    }

    @Override
    public void execute(OrderContext context) {
        OrderEntity order = context.getOrder();
        order.setAcceptTime(new Date());
        // 触发物流状态联动：商家接单后，开始呼叫骑手
        context.addPendingTransition(
                StatusDimension.LOGISTICS,
                LogisticsStatusEnum.WAIT_CALL_RIDER.getCode()
        );
    }
}
```

### 3.2 订单生命周期管理

#### 3.2.1 订单流转主线

订单的核心流转路径如下：

```
PRE_ACCEPT(1000) -> PRE_MAKE(1100) -> MAKING(1125) -> WAIT_DELIVERY(1150)
    -> DELIVERING(1200) -> COMPLETED(1400)
                        -> CANCELED(1500)
```

```java
/**
 * 订单生命周期流转码
 * 对外展示用的聚合状态码
 */
public enum OrderFlowCode {
    PRE_ACCEPT(1000, "待接单"),
    PRE_MAKE(1100, "待出餐"),
    MAKING(1125, "出餐中"),
    WAIT_DELIVERY(1150, "待配送"),
    DELIVERING(1200, "配送中"),
    COMPLETED(1400, "已完成"),
    CANCELED(1500, "已取消");

    private final int code;
    private final String desc;

    OrderFlowCode(int code, String desc) {
        this.code = code;
        this.desc = desc;
    }

    /**
     * 根据三维状态计算流转码
     * 优先级：退款 > 商家 > 物流
     */
    public static OrderFlowCode calculate(int merchantStatus,
                                           int logisticsStatus,
                                           int refundStatus) {
        // 退款状态优先
        if (refundStatus == RefundStatusEnum.REFUND_DONE.getCode()) {
            return CANCELED;
        }

        // 商家维度判断
        if (merchantStatus == MerchantStatusEnum.WAIT_ACCEPT.getCode()) {
            return PRE_ACCEPT;
        }
        if (merchantStatus == MerchantStatusEnum.ACCEPTED.getCode()
                || merchantStatus == MerchantStatusEnum.WAIT_PREPARE.getCode()) {
            return PRE_MAKE;
        }
        if (merchantStatus == MerchantStatusEnum.PREPARED.getCode()
                || merchantStatus == MerchantStatusEnum.PREPARE_TIMEOUT.getCode()) {
            // 已出餐后看物流状态
            if (logisticsStatus < LogisticsStatusEnum.RIDER_PICKED.getCode()) {
                return WAIT_DELIVERY;
            }
            return DELIVERING;
        }
        if (merchantStatus == MerchantStatusEnum.FULFILL_DONE.getCode()) {
            return COMPLETED;
        }
        if (merchantStatus == MerchantStatusEnum.REJECTED.getCode()) {
            return CANCELED;
        }

        return PRE_ACCEPT; // 默认
    }
}
```

#### 3.2.2 订单创建流程

```java
/**
 * 订单创建服务
 */
@Service
public class OrderCreateService {

    @Autowired
    private OrderIdGenerator orderIdGenerator;
    @Autowired
    private OrderRepository orderRepository;
    @Autowired
    private InventoryService inventoryService;
    @Autowired
    private CouponService couponService;
    @Autowired
    private PaymentService paymentService;
    @Autowired
    private OrderEventPublisher eventPublisher;

    /**
     * 创建订单 - 分布式事务保证
     */
    @Transactional(rollbackFor = Exception.class)
    public OrderCreateResult createOrder(OrderCreateRequest request) {
        // 1. 幂等性校验
        String idempotentKey = request.getIdempotentKey();
        if (orderRepository.existsByIdempotentKey(idempotentKey)) {
            OrderEntity existing = orderRepository.findByIdempotentKey(idempotentKey);
            return OrderCreateResult.duplicate(existing.getOrderId());
        }

        // 2. 生成订单ID
        long orderId = orderIdGenerator.generate();

        // 3. 锁定库存（预扣减）
        List<InventoryLockResult> lockResults = inventoryService.lockInventory(
                request.getMerchantId(),
                request.getItems().stream()
                        .map(item -> new InventoryLockItem(item.getSkuId(), item.getQuantity()))
                        .collect(Collectors.toList()),
                orderId
        );
        if (!lockResults.stream().allMatch(InventoryLockResult::isSuccess)) {
            throw new BusinessException("库存不足");
        }

        // 4. 锁定优惠券
        if (request.getCouponId() != null) {
            boolean couponLocked = couponService.lockCoupon(
                    request.getUserId(), request.getCouponId(), orderId);
            if (!couponLocked) {
                // 回滚库存
                inventoryService.unlockInventory(orderId);
                throw new BusinessException("优惠券不可用");
            }
        }

        // 5. 计算订单金额
        OrderAmount orderAmount = calculateOrderAmount(request);

        // 6. 创建订单实体
        OrderEntity order = new OrderEntity();
        order.setOrderId(orderId);
        order.setUserId(request.getUserId());
        order.setMerchantId(request.getMerchantId());
        order.setMerchantStatus(MerchantStatusEnum.INIT.getCode());
        order.setLogisticsStatus(LogisticsStatusEnum.INIT.getCode());
        order.setRefundStatus(RefundStatusEnum.NORMAL.getCode());
        order.setTotalAmount(orderAmount.getTotalAmount());
        order.setPayAmount(orderAmount.getPayAmount());
        order.setIdempotentKey(idempotentKey);
        order.setLastOrderStatus(0L);
        order.setCreateTime(new Date());

        orderRepository.save(order);

        // 7. 保存订单商品明细
        List<OrderItemEntity> items = buildOrderItems(orderId, request.getItems());
        orderRepository.saveItems(items);

        // 8. 发布订单创建事件
        eventPublisher.publish(new OrderCreatedEvent(orderId, request.getUserId(),
                request.getMerchantId()));

        return OrderCreateResult.success(orderId, orderAmount.getPayAmount());
    }

    /**
     * 计算订单金额
     */
    private OrderAmount calculateOrderAmount(OrderCreateRequest request) {
        BigDecimal totalAmount = BigDecimal.ZERO;
        for (OrderItemRequest item : request.getItems()) {
            totalAmount = totalAmount.add(
                    item.getPrice().multiply(BigDecimal.valueOf(item.getQuantity()))
            );
        }

        BigDecimal discountAmount = BigDecimal.ZERO;
        if (request.getCouponId() != null) {
            discountAmount = couponService.calculateDiscount(
                    request.getCouponId(), totalAmount);
        }

        BigDecimal deliveryFee = calculateDeliveryFee(request);
        BigDecimal payAmount = totalAmount.subtract(discountAmount).add(deliveryFee);
        if (payAmount.compareTo(BigDecimal.ZERO) < 0) {
            payAmount = BigDecimal.ZERO;
        }

        return new OrderAmount(totalAmount, discountAmount, deliveryFee, payAmount);
    }

    private BigDecimal calculateDeliveryFee(OrderCreateRequest request) {
        // 根据距离、时段、活动计算配送费
        return BigDecimal.valueOf(5.0);
    }
```

```java
/**
 * 订单ID生成器 - 雪花算法变体
 * 结构：1bit符号位 + 41bit时间戳 + 5bit数据中心 + 5bit机器 + 12bit序列号
 */
public class OrderIdGenerator {

    private static final long EPOCH = 1609459200000L; // 2021-01-01 00:00:00
    private static final long DATACENTER_BITS = 5L;
    private static final long WORKER_BITS = 5L;
    private static final long SEQUENCE_BITS = 12L;

    private static final long MAX_DATACENTER_ID = ~(-1L << DATACENTER_BITS);
    private static final long MAX_WORKER_ID = ~(-1L << WORKER_BITS);
    private static final long MAX_SEQUENCE = ~(-1L << SEQUENCE_BITS);

    private static final long WORKER_SHIFT = SEQUENCE_BITS;
    private static final long DATACENTER_SHIFT = SEQUENCE_BITS + WORKER_BITS;
    private static final long TIMESTAMP_SHIFT = SEQUENCE_BITS + WORKER_BITS + DATACENTER_BITS;

    private final long datacenterId;
    private final long workerId;
    private long sequence = 0L;
    private long lastTimestamp = -1L;

    public OrderIdGenerator(long datacenterId, long workerId) {
        if (datacenterId > MAX_DATACENTER_ID || datacenterId < 0) {
            throw new IllegalArgumentException("Datacenter ID out of range");
        }
        if (workerId > MAX_WORKER_ID || workerId < 0) {
            throw new IllegalArgumentException("Worker ID out of range");
        }
        this.datacenterId = datacenterId;
        this.workerId = workerId;
    }

    public synchronized long generate() {
        long timestamp = System.currentTimeMillis();
        if (timestamp < lastTimestamp) {
            throw new RuntimeException("Clock moved backwards");
        }
        if (timestamp == lastTimestamp) {
            sequence = (sequence + 1) & MAX_SEQUENCE;
            if (sequence == 0) {
                timestamp = waitNextMillis(lastTimestamp);
            }
        } else {
            sequence = 0L;
        }
        lastTimestamp = timestamp;

        return ((timestamp - EPOCH) << TIMESTAMP_SHIFT)
                | (datacenterId << DATACENTER_SHIFT)
                | (workerId << WORKER_SHIFT)
                | sequence;
    }

    private long waitNextMillis(long lastTimestamp) {
        long timestamp = System.currentTimeMillis();
        while (timestamp <= lastTimestamp) {
            timestamp = System.currentTimeMillis();
        }
        return timestamp;
    }
}
```

```java
/**
```

```java
@Service
public class RefundService {

    @Autowired
    private OrderStateMachine stateMachine;
    @Autowired
    private OrderRepository orderRepository;
    @Autowired
    private PaymentService paymentService;
    @Autowired
    private InventoryService inventoryService;
    @Autowired
    private CouponService couponService;

    /**
     * 申请退款
     */
    @Transactional(rollbackFor = Exception.class)
    public RefundApplyResult applyRefund(RefundApplyRequest request) {
        long orderId = request.getOrderId();
        OrderEntity order = orderRepository.findById(orderId);

        // 1. 校验退款条件
        validateRefundCondition(order, request);

        // 2. 更新退款状态
        TransitionResult result = stateMachine.transition(
                new OrderContext(order),
                StatusDimension.REFUND,
                RefundStatusEnum.WAIT_APPROVE.getCode()
        );

        if (!result.isSuccess()) {
            return RefundApplyResult.fail(result.getMessage());
        }

        // 3. 创建退款单
        RefundOrder refundOrder = new RefundOrder();
        refundOrder.setOrderId(orderId);
        refundOrder.setRefundAmount(request.getRefundAmount());
        refundOrder.setReason(request.getReason());
        refundOrder.setType(request.getRefundType());
        refundOrder.setStatus(RefundOrderStatus.PENDING);
        refundOrder.setCreateTime(new Date());
        refundOrderRepository.save(refundOrder);

        // 4. 判断是否需要自动审批
        if (canAutoApprove(order, request)) {
            return approveRefund(refundOrder.getRefundId());
        }

        return RefundApplyResult.success(refundOrder.getRefundId());
    }

    /**
     * 审批退款
     */
    @Transactional(rollbackFor = Exception.class)
    public RefundApplyResult approveRefund(long refundId) {
        RefundOrder refundOrder = refundOrderRepository.findById(refundId);
        OrderEntity order = orderRepository.findById(refundOrder.getOrderId());

        // 1. 更新退款状态为退款中
        stateMachine.transition(
                new OrderContext(order),
                StatusDimension.REFUND,
                RefundStatusEnum.REFUNDING.getCode()
        );

        // 2. 调用支付退款
        PaymentRefundResult payResult = paymentService.refund(
                order.getOrderId(),
                refundOrder.getRefundAmount()
        );

        if (!payResult.isSuccess()) {
            // 退款失败，发起重试
            retryRefund(refundId);
            return RefundApplyResult.fail("退款处理中");
        }

        // 3. 退款成功后处理
        handleRefundSuccess(order, refundOrder);

        return RefundApplyResult.success(refundId);
    }

    /**
     * 退款成功后处理
     */
    private void handleRefundSuccess(OrderEntity order, RefundOrder refundOrder) {
        // 1. 更新退款状态
        stateMachine.transition(
                new OrderContext(order),
                StatusDimension.REFUND,
                RefundStatusEnum.REFUND_DONE.getCode()
        );

        // 2. 释放库存
        inventoryService.releaseInventory(order.getOrderId());

        // 3. 退还优惠券
        couponService.returnCoupon(order.getOrderId());

        // 4. 取消配送
        if (order.getLogisticsStatus() < LogisticsStatusEnum.RIDER_DELIVERED.getCode()) {
            deliveryService.cancelDelivery(order.getOrderId(), "订单退款");
        }
    }

    /**
     * 校验退款条件
     */
    private void validateRefundCondition(OrderEntity order, RefundApplyRequest request) {
        // 已完成退款的订单不能重复退款
        if (order.getRefundStatus() == RefundStatusEnum.REFUND_DONE.getCode()) {
            throw new BusinessException("订单已退款完成");
        }
        // 正在退款中的订单不能重复申请
        if (order.getRefundStatus() == RefundStatusEnum.REFUNDING.getCode()
                || order.getRefundStatus() == RefundStatusEnum.WAIT_APPROVE.getCode()) {
            throw new BusinessException("订单退款处理中");
        }
        // 退款金额校验
        if (request.getRefundAmount().compareTo(order.getPayAmount()) > 0) {
            throw new BusinessException("退款金额不能超过实付金额");
        }
    }

    /**
     * 判断是否自动审批
     */
    private boolean canAutoApprove(OrderEntity order, RefundApplyRequest request) {
        // 商家未接单时自动审批
        if (order.getMerchantStatus() <= MerchantStatusEnum.WAIT_ACCEPT.getCode()) {
            return true;
        }
        // 小额退款自动审批
        if (request.getRefundAmount().compareTo(new BigDecimal("10")) <= 0) {
            return true;
        }
        return false;
    }

    /**
     * 退款重试
     */
    private void retryRefund(long refundId) {
        scheduledExecutor.schedule(() -> {
            try {
                approveRefund(refundId);
            } catch (Exception e) {
                log.error("退款重试失败, refundId={}", refundId, e);
                // 达到最大重试次数后告警
            }
        }, 30, TimeUnit.SECONDS);
    }
}
```

```java
@Service
public class OrderPushNotificationService {

    static {

        PUSH_CONFIGS.put(MerchantStatusEnum.ACCEPTED.getCode(),
                new PushConfig(true, false, false)); // 仅推送
        PUSH_CONFIGS.put(LogisticsStatusEnum.RIDER_PICKED.getCode(),
                new PushConfig(true, false, false)); // 仅推送
        PUSH_CONFIGS.put(LogisticsStatusEnum.RIDER_DELIVERED.getCode(),

     * 状态变更监听 - 触发推送

    public void onStatusChanged(StatusChangeEvent event) {

        int newStatus = event.getToStatus();
        long lastOrderStatus = event.getLastOrderStatus();

        // 构建当前状态快照
        long currentStatusSnapshot = buildStatusSnapshot(event);

        // 防重复推送：比较 lastOrderStatus
        if (currentStatusSnapshot == lastOrderStatus) {
            log.info("跳过重复推送, orderId={}, status={}", orderId, newStatus);
            return;
        }

        // 更新 lastOrderStatus
        orderRepository.updateLastOrderStatus(orderId, currentStatusSnapshot);

        // 查找推送配置
        PushConfig config = PUSH_CONFIGS.get(newStatus);
        if (config == null) {
            return;
        }

        // 执行推送
        if (config.needPush()) {
            pushService.pushOrderStatusChange(orderId, newStatus);
        }
        if (config.needVoice()) {
            voiceCallService.notifyMerchant(orderId);
        }
        if (config.needIvr()) {
            ivrService.callMerchant(orderId);
        }
    }

    /**
     * 构建状态快照
     * 将三维状态编码为一个 long 值
     */
    private long buildStatusSnapshot(StatusChangeEvent event) {
        OrderEntity order = orderRepository.findById(event.getOrderId());
        return ((long) order.getMerchantStatus() << 32)
                | ((long) order.getLogisticsStatus() << 16)
                | order.getRefundStatus();
    }
}

/**
```

```java
@Service
public class IdempotencyService {

    @Autowired
    private RedisTemplate<String, String> redisTemplate;

    private static final String IDEMPOTENT_PREFIX = "order:idempotent:";
    private static final long DEFAULT_EXPIRE_SECONDS = 86400; // 24小时

    /**
     * 幂等性校验并执行
     */
    public <T> T executeIdempotent(String idempotentKey,
                                    Supplier<T> action,
                                    Function<String, T> onDuplicate) {
        String key = IDEMPOTENT_PREFIX + idempotentKey;

        // 1. 尝试设置幂等键
        Boolean success = redisTemplate.opsForValue()
                .setIfAbsent(key, "PROCESSING", DEFAULT_EXPIRE_SECONDS, TimeUnit.SECONDS);

        if (Boolean.FALSE.equals(success)) {
            // 重复请求
            String result = redisTemplate.opsForValue().get(key);
            return onDuplicate.apply(result);
        }

        try {
            // 2. 执行业务逻辑
            T result = action.get();

            // 3. 更新幂等键的值为结果
            redisTemplate.opsForValue().set(key, JSON.toJSONString(result),
                    DEFAULT_EXPIRE_SECONDS, TimeUnit.SECONDS);

            return result;
        } catch (Exception e) {
            // 执行失败，删除幂等键，允许重试
            redisTemplate.delete(key);
            throw e;
        }
    }

    /**
     * 订单状态变更幂等
     */
    public boolean tryTransition(long orderId, StatusDimension dimension,
                                  int fromStatus, int toStatus) {
        String key = String.format("order:transition:%d:%s:%d:%d",
                orderId, dimension.name(), fromStatus, toStatus);

        Boolean success = redisTemplate.opsForValue()
                .setIfAbsent(key, "1", 3600, TimeUnit.SECONDS);
        return Boolean.TRUE.equals(success);
    }
}
```

```java
/**
 * 订单补偿服务 - 基于 TCC 模式
 */
@Service
public class OrderCompensationService {

    @Autowired
    private CompensationLogRepository compensationLogRepo;

    /**
     * TCC Try 阶段 - 资源预留
     */
    public TryResult tryPhase(OrderCreateRequest request) {
        TryResult result = new TryResult();

        // 1. 库存预留
        boolean inventoryReserved = inventoryService.tryReserve(
                request.getMerchantId(), request.getItems());
        result.setInventoryReserved(inventoryReserved);
        if (!inventoryReserved) {
            return result;
        }

        // 2. 优惠券预锁
        if (request.getCouponId() != null) {
            boolean couponLocked = couponService.tryLock(
                    request.getUserId(), request.getCouponId());
            result.setCouponLocked(couponLocked);
            if (!couponLocked) {
                // 回滚库存
                inventoryService.cancelReserve(request.getMerchantId(), request.getItems());
                return result;
            }
        }

        result.setSuccess(true);
        return result;
    }

    /**
     * TCC Confirm 阶段 - 确认提交
     */
    public void confirmPhase(long orderId) {
        CompensationLog log = compensationLogRepo.findByOrderId(orderId);

        if (log.isInventoryReserved()) {
            inventoryService.confirmReserve(orderId);
        }
        if (log.isCouponLocked()) {
            couponService.confirmLock(orderId);
        }

        log.setPhase("CONFIRMED");
        compensationLogRepo.save(log);
    }

    /**
     * TCC Cancel 阶段 - 取消回滚
     */
    public void cancelPhase(long orderId) {
        CompensationLog log = compensationLogRepo.findByOrderId(orderId);

        if (log.isInventoryReserved()) {
            inventoryService.cancelReserve(orderId);
        }
        if (log.isCouponLocked()) {
            couponService.cancelLock(orderId);
        }

        log.setPhase("CANCELLED");
        compensationLogRepo.save(log);
    }
}
```

```java
                .findByMerchantStatusAndCreateTimeBefore(
```

```java
@Service
public class OrderConsistencyChecker {

    /**
     * 定时对账：订单与支付
     */
    @Scheduled(cron = "0 0 2 * * ?") // 每天凌晨2点
    public void reconcileOrderPayment() {
        Date yesterday = DateUtils.addDays(new Date(), -1);
        List<OrderEntity> orders = orderRepository.findByCreateDate(yesterday);
        List<PaymentRecord> payments = paymentService.findByDate(yesterday);

        Map<Long, PaymentRecord> paymentMap = payments.stream()
                .collect(Collectors.toMap(PaymentRecord::getOrderId, p -> p));

        for (OrderEntity order : orders) {
            PaymentRecord payment = paymentMap.get(order.getOrderId());

            // 检查1：有订单无支付记录
            if (payment == null && order.getMerchantStatus() > MerchantStatusEnum.INIT.getCode()) {
                reportInconsistency(order.getOrderId(), "订单已支付但无支付记录");
                continue;
            }

            // 检查2：金额不一致
            if (payment != null && !payment.getAmount().equals(order.getPayAmount())) {
                reportInconsistency(order.getOrderId(),
                        String.format("金额不一致: 订单=%s, 支付=%s",
                                order.getPayAmount(), payment.getAmount()));
            }

            // 检查3：状态不一致
            if (payment != null && payment.getStatus() == PaymentStatus.SUCCESS
                    && order.getMerchantStatus() == MerchantStatusEnum.INIT.getCode()) {
                reportInconsistency(order.getOrderId(), "支付成功但订单未更新");
                // 自动修复
                autoFixPaymentStatus(order, payment);
            }
        }
    }

    private void reportInconsistency(long orderId, String message) {
        log.error("数据一致性异常: orderId={}, message={}", orderId, message);
        alertService.sendAlert(AlertLevel.HIGH, "数据一致性异常: " + message);
    }

    private void autoFixPaymentStatus(OrderEntity order, PaymentRecord payment) {
        log.warn("自动修复订单状态, orderId={}", order.getOrderId());
        stateMachine.transition(
                new OrderContext(order),
                StatusDimension.MERCHANT,
                MerchantStatusEnum.WAIT_ACCEPT.getCode()
        );
    }
}
```

```java
@Service
public class OrderCacheService {

    private static final String ORDER_CACHE_PREFIX = "order:detail:";
    private static final int ACTIVE_ORDER_EXPIRE_SECONDS = 3600; // 1小时
    private static final int COMPLETED_ORDER_EXPIRE_SECONDS = 300; // 5分钟

    @Autowired
    private RedisTemplate<String, String> redisTemplate;
    @Autowired
    private OrderRepository orderRepository;

    /**
     * 读取订单 - 缓存优先
     */
    public OrderEntity getOrder(long orderId) {
        String key = ORDER_CACHE_PREFIX + orderId;

        // 1. 读缓存
        String cached = redisTemplate.opsForValue().get(key);
        if (cached != null) {
            return JSON.parseObject(cached, OrderEntity.class);
        }

        // 2. 读数据库
        OrderEntity order = orderRepository.findById(orderId);
        if (order == null) {
            return null;
        }

        // 3. 写入缓存
        int expireSeconds = isActiveOrder(order)
                ? ACTIVE_ORDER_EXPIRE_SECONDS
                : COMPLETED_ORDER_EXPIRE_SECONDS;
        redisTemplate.opsForValue().set(key, JSON.toJSONString(order),
                expireSeconds, TimeUnit.SECONDS);

        return order;
    }

    /**
     * 更新订单缓存
     */
    public void updateCache(OrderEntity order) {
        String key = ORDER_CACHE_PREFIX + order.getOrderId();
        int expireSeconds = isActiveOrder(order)
                ? ACTIVE_ORDER_EXPIRE_SECONDS
                : COMPLETED_ORDER_EXPIRE_SECONDS;
        redisTemplate.opsForValue().set(key, JSON.toJSONString(order),
                expireSeconds, TimeUnit.SECONDS);
    }

    /**
     * 删除缓存 - 状态变更时
     */
    public void invalidateCache(long orderId) {

        redisTemplate.delete(key);
```

```java
@Service
public class OrderListAggregator {

    @Autowired
    private ExecutorService orderExecutor;

    /**
     * 聚合查询用户订单列表
     */
    public OrderListDTO aggregateOrderList(long userId, int page, int size) {
        // 1. 查询订单基础信息
        CompletableFuture<List<OrderEntity>> ordersFuture = CompletableFuture
                .supplyAsync(() -> orderRepository.findByUserId(userId, page, size),
                        orderExecutor);

        List<OrderEntity> orders;
        try {
            orders = ordersFuture.get(3, TimeUnit.SECONDS);
        } catch (Exception e) {
            log.error("查询订单列表超时, userId={}", userId, e);
            return OrderListDTO.empty();
        }

        if (orders.isEmpty()) {
            return OrderListDTO.empty();
        }

        List<Long> orderIds = orders.stream()
                .map(OrderEntity::getOrderId)
                .collect(Collectors.toList());
        List<Long> merchantIds = orders.stream()
                .map(OrderEntity::getMerchantId)
                .distinct()
                .collect(Collectors.toList());

        // 2. 并行查询关联数据
        CompletableFuture<Map<Long, MerchantInfo>> merchantsFuture = CompletableFuture
                .supplyAsync(() -> merchantService.batchGet(merchantIds), orderExecutor);

        CompletableFuture<Map<Long, List<OrderItemEntity>>> itemsFuture = CompletableFuture
                .supplyAsync(() -> orderRepository.batchGetItems(orderIds), orderExecutor);

        CompletableFuture<Map<Long, DeliveryInfo>> deliveryFuture = CompletableFuture
                .supplyAsync(() -> deliveryService.batchGetInfo(orderIds), orderExecutor);

        try {
            // 等待所有结果
            CompletableFuture.allOf(merchantsFuture, itemsFuture, deliveryFuture)
                    .get(5, TimeUnit.SECONDS);

            Map<Long, MerchantInfo> merchants = merchantsFuture.get();
            Map<Long, List<OrderItemEntity>> items = itemsFuture.get();
            Map<Long, DeliveryInfo> deliveries = deliveryFuture.get();

            // 3. 聚合结果
            List<OrderDetailDTO> details = orders.stream().map(order -> {
                OrderDetailDTO dto = new OrderDetailDTO();
                dto.setOrder(order);
                dto.setMerchant(merchants.get(order.getMerchantId()));
                dto.setItems(items.getOrDefault(order.getOrderId(), Collections.emptyList()));
                dto.setDelivery(deliveries.get(order.getOrderId()));
                return dto;
            }).collect(Collectors.toList());

            return new OrderListDTO(details, orderRepository.countByUserId(userId));
        } catch (Exception e) {
            log.error("聚合订单数据超时, userId={}", userId, e);
            // 降级：只返回订单基础信息
            return OrderListDTO.degraded(orders);
        }
    }
```

## 六、最佳实践

### 6.1 订单系统设计原则

1. **状态机驱动**：所有状态变更通过状态机统一管控，避免散乱的 if-else 判断。状态机需要校验合法性、记录变更日志、触发事件。

2. **多维度状态分离**：将不同业务线的状态独立管理，避免单一状态码的爆炸增长。每个维度可以独立演进。

3. **幂等性设计**：所有状态变更接口必须具备幂等性。使用唯一键 + Redis 原子操作保障。

4. **补偿机制**：分布式环境下，网络故障不可避免。必须设计重试和补偿机制，确保最终一致性。

5. **推送防重复**：使用 lastOrderStatus 快照机制，避免状态变更风暴时重复推送通知。

### 6.4 关键注意事项

1. **订单号全局唯一**：使用雪花算法或类似方案，确保在分布式环境下订单号不重复。

2. **状态机要支持并发**：多个事件可能同时到达，状态机需要加锁或使用乐观锁保证线程安全。

3. **老系统兼容**：在迁移过程中，搜索接口需同时支持新旧状态码，逐步下线旧接口。

4. **订单数据归档**：超过一定时间的订单移到历史库，减轻在线库压力。建议完成30天以上的订单执行归档。

5. **灰度发布**：新旧状态体系切换需要灰度，按商家维度灰度，先小范围验证再全量切换。

6. **降级策略**：
   - 缓存降级：Redis不可用时直接读数据库
   - 聚合降级：关联数据查询超时时只返回订单基础信息
   - 推送降级：推送通道不可用时降级为短信通知

7. **日志规范**：所有状态变更必须记录完整日志（订单号、维度、原状态、目标状态、操作人、时间），便于问题排查和数据对账。

### 7.1 案例一：用户下单全链路

#### 7.1.1 链路概述

用户从提交订单到支付完成的完整链路，涉及参数校验、库存预扣、优惠计算、订单入库、支付跳转、支付回调、订单状态更新等多个环节。每个环节都需要保证幂等性和异常可恢复性。

```
提交订单 -> 参数校验 -> 库存预扣 -> 优惠计算 -> 订单入库 -> 支付跳转 -> 支付回调 -> 订单状态更新
    |          |          |          |          |          |          |          |
    |  幂等键校验  |  Redis原子扣  |  优惠券锁定  |  分布式事务  |  支付单创建  |  签名验证  |  状态机流转  |
    |  参数完整性  |  失败回滚    |  失败回滚    |  事件发布    |  SDK跳转    |  幂等处理  |  缓存失效    |
```

#### 7.1.2 完整代码实现

```java
/**
 * 用户下单全链路服务
 * 涵盖：参数校验 -> 库存预扣 -> 优惠计算 -> 订单入库 -> 支付跳转 -> 支付回调 -> 订单状态更新
 */
@Service
@Slf4j
public class OrderFullChainService {

    @Autowired
    private OrderRepository orderRepository;
    @Autowired
    private OrderIdGenerator orderIdGenerator;
    @Autowired
    private InventoryService inventoryService;
    @Autowired
    private CouponService couponService;
    @Autowired
    private PaymentService paymentService;
    @Autowired
    private IdempotencyService idempotencyService;
    @Autowired
    private OrderStateMachine stateMachine;
    @Autowired
    private OrderCacheService orderCacheService;
    @Autowired
    private OrderEventPublisher eventPublisher;
    @Autowired
    private OrderMetrics orderMetrics;

    private static final String ORDER_CREATE_IDEMPOTENT_PREFIX = "order:create:";
    private static final String PAY_CALLBACK_IDEMPOTENT_PREFIX = "order:pay:callback:";
    private static final long IDEMPOTENT_EXPIRE_SECONDS = 86400L;

    // ==================== 第一步：提交订单（入口） ====================

    /**
     * 提交订单 - 全链路入口
     * 幂等控制：基于客户端传入的 idempotentKey 防重复提交
     */
    public OrderSubmitResult submitOrder(OrderSubmitRequest request) {
        String idempotentKey = request.getIdempotentKey();
        if (StringUtils.isBlank(idempotentKey)) {
            log.warn("提交订单失败，幂等键为空, userId={}", request.getUserId());
            throw new BusinessException("幂等键不能为空");
        }

        log.info("提交订单开始, userId={}, merchantId={}, idempotentKey={}",
                request.getUserId(), request.getMerchantId(), idempotentKey);

        try {
            return idempotencyService.executeIdempotent(
                    ORDER_CREATE_IDEMPOTENT_PREFIX + idempotentKey,
                    () -> doSubmitOrder(request),
                    (existingResult) -> {
                        log.info("重复提交订单请求, idempotentKey={}, 返回已有结果", idempotentKey);
                        OrderSubmitResult result = JSON.parseObject(existingResult, OrderSubmitResult.class);
                        result.setDuplicate(true);
                        return result;
                    }
            );
        } catch (BusinessException e) {
            log.warn("提交订单业务异常, userId={}, error={}", request.getUserId(), e.getMessage());
            throw e;
        } catch (Exception e) {
            log.error("提交订单系统异常, userId={}", request.getUserId(), e);
            throw new SystemException("订单提交失败，请稍后重试");
        }
    }

    /**
     * 执行提交订单核心逻辑
     */
    @Transactional(rollbackFor = Exception.class)
    protected OrderSubmitResult doSubmitOrder(OrderSubmitRequest request) {
        // 1. 参数校验
        validateOrderRequest(request);
        log.info("参数校验通过, userId={}, merchantId={}", request.getUserId(), request.getMerchantId());

        // 2. 生成订单ID
        long orderId = orderIdGenerator.generate();
        log.info("生成订单ID, orderId={}", orderId);

        // 3. 库存预扣（Redis 原子扣减 + DB 记录）
        List<InventoryLockItem> lockItems = request.getItems().stream()
                .map(item -> new InventoryLockItem(item.getSkuId(), item.getQuantity()))
                .collect(Collectors.toList());
        InventoryLockResult lockResult = inventoryService.lockInventory(
                request.getMerchantId(), lockItems, orderId);
        if (!lockResult.isAllSuccess()) {
            log.warn("库存预扣失败, orderId={}, failedSkus={}", orderId, lockResult.getFailedSkus());
            throw new BusinessException("库存不足: " + lockResult.getFailedSkus());
        }
        log.info("库存预扣成功, orderId={}", orderId);

        // 4. 优惠计算 + 优惠券锁定
        OrderAmount orderAmount;
        if (request.getCouponId() != null) {
            boolean couponLocked = couponService.lockCoupon(
                    request.getUserId(), request.getCouponId(), orderId);
            if (!couponLocked) {
                // 回滚库存
                inventoryService.unlockInventory(orderId);
                log.warn("优惠券锁定失败, orderId={}, couponId={}", orderId, request.getCouponId());
                throw new BusinessException("优惠券不可用或已被使用");
            }
            BigDecimal discount = couponService.calculateDiscount(
                    request.getCouponId(), calculateTotalAmount(request));
            orderAmount = new OrderAmount(
                    calculateTotalAmount(request),
                    discount,
                    calculateDeliveryFee(request),
                    calculateTotalAmount(request).subtract(discount).add(calculateDeliveryFee(request))
            );
            log.info("优惠券计算完成, orderId={}, discount={}", orderId, discount);
        } else {
            orderAmount = new OrderAmount(
                    calculateTotalAmount(request),
                    BigDecimal.ZERO,
                    calculateDeliveryFee(request),
                    calculateTotalAmount(request).add(calculateDeliveryFee(request))
            );
        }

        // 5. 订单入库
        OrderEntity order = buildOrderEntity(orderId, request, orderAmount);
        orderRepository.save(order);
        orderRepository.saveItems(buildOrderItems(orderId, request.getItems()));
        log.info("订单入库成功, orderId={}, payAmount={}", orderId, orderAmount.getPayAmount());

        // 6. 状态机流转：INIT -> WAIT_ACCEPT
        OrderContext context = new OrderContext(order);
        TransitionResult transitionResult = stateMachine.transition(
                context,
                StatusDimension.MERCHANT,
                MerchantStatusEnum.WAIT_ACCEPT.getCode()
        );
        if (!transitionResult.isSuccess()) {
            log.error("订单状态流转失败, orderId={}, reason={}", orderId, transitionResult.getMessage());
            throw new SystemException("订单状态初始化失败");
        }

        // 7. 发布订单创建事件
        eventPublisher.publish(new OrderCreatedEvent(orderId, request.getUserId(),
                request.getMerchantId()));

        // 8. 创建支付单并返回支付跳转信息
        PaymentCreateResult payResult = paymentService.createPayment(
                orderId, orderAmount.getPayAmount(), request.getPayChannel());
        if (!payResult.isSuccess()) {
            log.error("支付单创建失败, orderId={}", orderId);
            throw new SystemException("支付单创建失败");
        }
        log.info("支付单创建成功, orderId={}, payOrderId={}", orderId, payResult.getPayOrderId());

        // 9. 记录监控指标
        orderMetrics.recordOrderCreate();

        // 10. 返回结果
        OrderSubmitResult result = new OrderSubmitResult();
        result.setOrderId(orderId);
        result.setPayOrderId(payResult.getPayOrderId());
        result.setPayUrl(payResult.getPayUrl());
        result.setPayAmount(orderAmount.getPayAmount());
        result.setSuccess(true);

        log.info("提交订单完成, orderId={}, userId={}, payAmount={}",
                orderId, request.getUserId(), orderAmount.getPayAmount());
        return result;
    }

    // ==================== 第二步：支付回调 ====================

    /**
     * 支付回调处理
     * 幂等控制：基于支付流水号防重复回调
     */
    @Transactional(rollbackFor = Exception.class)
    public PaymentCallbackResult handlePaymentCallback(PaymentCallbackRequest request) {
        String payOrderId = request.getPayOrderId();
        long orderId = request.getOrderId();

        // 1. 签名校验
        if (!paymentService.verifyCallbackSignature(request)) {
            log.warn("支付回调签名校验失败, orderId={}, payOrderId={}", orderId, payOrderId);
            throw new BusinessException("签名校验失败");
        }

        // 2. 幂等控制：检查是否已处理过该回调
        String idempotentKey = PAY_CALLBACK_IDEMPOTENT_PREFIX + payOrderId;
        Boolean firstProcess = idempotencyService.tryLock(idempotentKey, IDEMPOTENT_EXPIRE_SECONDS);
        if (Boolean.FALSE.equals(firstProcess)) {
            log.info("支付回调重复请求, orderId={}, payOrderId={}", orderId, payOrderId);
            PaymentCallbackResult result = new PaymentCallbackResult();
            result.setSuccess(true);
            result.setDuplicate(true);
            result.setMessage("回调已处理");
            return result;
        }

        try {
            log.info("处理支付回调开始, orderId={}, payOrderId={}, status={}",
                    orderId, payOrderId, request.getPayStatus());

            // 3. 查询订单
            OrderEntity order = orderRepository.findById(orderId);
            if (order == null) {
                log.error("支付回调订单不存在, orderId={}", orderId);
                throw new BusinessException("订单不存在");
            }

            // 4. 校验支付状态
            if (request.getPayStatus() != PaymentStatus.SUCCESS) {
                log.warn("支付未成功, orderId={}, payStatus={}", orderId, request.getPayStatus());
                handlePaymentFailed(order, request);
                return PaymentCallbackResult.fail("支付未成功");
            }

            // 5. 校验金额一致性
            if (order.getPayAmount().compareTo(request.getPayAmount()) != 0) {
                log.error("支付金额不一致, orderId={}, orderAmount={}, payAmount={}",
                        orderId, order.getPayAmount(), request.getPayAmount());
                throw new SystemException("支付金额不一致");
            }

            // 6. 状态机流转：更新订单状态
            // 支付成功后，商家状态从 INIT/WAIT_ACCEPT 推进
            OrderContext context = new OrderContext(order);
            if (order.getMerchantStatus() == MerchantStatusEnum.INIT.getCode()) {
                // 未进入待接单状态，先推进到待接单
                stateMachine.transition(context, StatusDimension.MERCHANT,
                        MerchantStatusEnum.WAIT_ACCEPT.getCode());
            }

            // 7. 更新支付状态
            order.setPayStatus(PaymentStatus.SUCCESS);
            order.setPayTime(new Date());
            order.setPayOrderId(payOrderId);
            orderRepository.update(order);

            // 8. 缓存失效
            orderCacheService.invalidateCache(orderId);

            // 9. 发布支付成功事件
            eventPublisher.publish(new PaymentSuccessEvent(orderId, order.getUserId(),
                    payOrderId, order.getPayAmount()));

            log.info("支付回调处理完成, orderId={}, payOrderId={}", orderId, payOrderId);

            PaymentCallbackResult result = new PaymentCallbackResult();
            result.setSuccess(true);
            result.setOrderId(orderId);
            return result;

        } catch (BusinessException e) {
            log.warn("支付回调业务异常, orderId={}, error={}", orderId, e.getMessage());
            idempotencyService.releaseLock(idempotentKey);
            throw e;
        } catch (Exception e) {
            log.error("支付回调系统异常, orderId={}", orderId, e);
            idempotencyService.releaseLock(idempotentKey);
            throw new SystemException("支付回调处理失败");
        }
    }

    // ==================== 辅助方法 ====================

    /**
     * 参数校验
     */
    private void validateOrderRequest(OrderSubmitRequest request) {
        if (request.getUserId() == null || request.getUserId() <= 0) {
            throw new BusinessException("用户ID无效");
        }
        if (request.getMerchantId() == null || request.getMerchantId() <= 0) {
            throw new BusinessException("商家ID无效");
        }
        if (CollectionUtils.isEmpty(request.getItems())) {
            throw new BusinessException("订单商品不能为空");
        }
        for (OrderItemRequest item : request.getItems()) {
            if (item.getQuantity() <= 0) {
                throw new BusinessException("商品数量必须大于0, skuId=" + item.getSkuId());
            }
            if (item.getPrice() == null || item.getPrice().compareTo(BigDecimal.ZERO) < 0) {
                throw new BusinessException("商品价格无效, skuId=" + item.getSkuId());
            }
        }
        // 防止下单数量异常
        int totalQuantity = request.getItems().stream()
                .mapToInt(OrderItemRequest::getQuantity).sum();
        if (totalQuantity > 99) {
            throw new BusinessException("单次下单商品数量不能超过99");
        }
    }

    /**
     * 处理支付失败
     */
    private void handlePaymentFailed(OrderEntity order, PaymentCallbackRequest request) {
        log.warn("支付失败处理, orderId={}, reason={}", order.getOrderId(), request.getFailReason());
        // 发布支付失败事件，由超时服务处理关单
        eventPublisher.publish(new PaymentFailedEvent(
                order.getOrderId(), order.getUserId(), request.getFailReason()));
    }

    private BigDecimal calculateTotalAmount(OrderSubmitRequest request) {
        return request.getItems().stream()
                .map(item -> item.getPrice().multiply(BigDecimal.valueOf(item.getQuantity())))
                .reduce(BigDecimal.ZERO, BigDecimal::add);
    }

    private BigDecimal calculateDeliveryFee(OrderSubmitRequest request) {
        return BigDecimal.valueOf(5.0);
    }

    private OrderEntity buildOrderEntity(long orderId, OrderSubmitRequest request,
                                          OrderAmount orderAmount) {
        OrderEntity order = new OrderEntity();
        order.setOrderId(orderId);
        order.setUserId(request.getUserId());
        order.setMerchantId(request.getMerchantId());
        order.setMerchantStatus(MerchantStatusEnum.INIT.getCode());
        order.setLogisticsStatus(LogisticsStatusEnum.INIT.getCode());
        order.setRefundStatus(RefundStatusEnum.NORMAL.getCode());
        order.setTotalAmount(orderAmount.getTotalAmount());
        order.setPayAmount(orderAmount.getPayAmount());
        order.setIdempotentKey(request.getIdempotentKey());
        order.setLastOrderStatus(0L);
        order.setCreateTime(new Date());
        order.setUpdateTime(new Date());
        return order;
    }

    private List<OrderItemEntity> buildOrderItems(long orderId, List<OrderItemRequest> items) {
        return items.stream().map(item -> {
            OrderItemEntity entity = new OrderItemEntity();
            entity.setOrderId(orderId);
            entity.setSkuId(item.getSkuId());
            entity.setSkuName(item.getSkuName());
            entity.setPrice(item.getPrice());
            entity.setQuantity(item.getQuantity());
            return entity;
        }).collect(Collectors.toList());
    }
}

/**
 * 支付服务 - 支付单创建与回调签名校验
 */
@Service
@Slf4j
class PaymentService {

    @Autowired
    private PaymentRepository paymentRepository;

    private static final String SIGN_KEY = "payment_sign_secret_key";

    /**
     * 创建支付单
     */
    public PaymentCreateResult createPayment(long orderId, BigDecimal amount, String payChannel) {
        try {
            String payOrderId = "PAY" + System.currentTimeMillis() + orderId;
            PaymentEntity payment = new PaymentEntity();
            payment.setPayOrderId(payOrderId);
            payment.setOrderId(orderId);
            payment.setAmount(amount);
            payment.setChannel(payChannel);
            payment.setStatus(PaymentStatus.PENDING);
            payment.setCreateTime(new Date());
            paymentRepository.save(payment);

            // 调用第三方支付SDK获取支付链接
            String payUrl = callPaymentSDK(payOrderId, amount, payChannel);

            PaymentCreateResult result = new PaymentCreateResult();
            result.setSuccess(true);
            result.setPayOrderId(payOrderId);
            result.setPayUrl(payUrl);
            return result;
        } catch (Exception e) {
            log.error("创建支付单异常, orderId={}", orderId, e);
            return PaymentCreateResult.fail("支付单创建失败");
        }
    }

    /**
     * 回调签名校验
     */
    public boolean verifyCallbackSignature(PaymentCallbackRequest request) {
        String expectedSign = DigestUtils.md5Hex(
                request.getPayOrderId() + request.getOrderId()
                        + request.getPayAmount() + SIGN_KEY);
        return expectedSign.equals(request.getSign());
    }

    private String callPaymentSDK(String payOrderId, BigDecimal amount, String channel) {
        // 实际调用微信/支付宝SDK
        return "https://pay.example.com/" + channel + "?orderId=" + payOrderId;
    }
}
```

### 7.2 案例二：订单超时取消全链路

#### 7.2.1 链路概述

订单创建后若用户未在规定时间内完成支付，系统需自动取消订单并释放占用的资源（库存、优惠券），同时通知用户。本案例提供延迟消息和定时扫描双保险机制。

```
延迟消息/定时扫描 -> 订单状态校验 -> 库存释放 -> 优惠券退回 -> 通知用户
      |                 |              |             |            |
      |  RocketMQ延迟   |  幂等校验     |  DB+Redis   |  优惠券   |  推送+短信 |
      |  定时扫描兜底   |  CAS乐观锁    |  双写释放   |  状态回退  |  多通道通知 |
```

#### 7.2.2 完整代码实现

```java
/**
 * 订单超时取消全链路服务
 * 涵盖：延迟消息/定时扫描 -> 订单状态校验 -> 库存释放 -> 优惠券退回 -> 通知用户
 */
@Service
@Slf4j
public class OrderTimeoutCancelService {

    @Autowired
    private OrderRepository orderRepository;
    @Autowired
    private InventoryService inventoryService;
    @Autowired
    private CouponService couponService;
    @Autowired
    private IdempotencyService idempotencyService;
    @Autowired
    private OrderStateMachine stateMachine;
    @Autowired
    private OrderCacheService orderCacheService;
    @Autowired
    private PushService pushService;
    @Autowired
    private SmsService smsService;
    @Autowired
    private OrderEventPublisher eventPublisher;
    @Autowired
    private OrderMetrics orderMetrics;

    private static final String CANCEL_IDEMPOTENT_PREFIX = "order:cancel:timeout:";
    private static final long IDEMPOTENT_EXPIRE_SECONDS = 86400L;
    private static final int PAY_TIMEOUT_MINUTES = 15;

    // ==================== 方式一：延迟消息触发 ====================

    /**
     * 延迟消息消费者 - 处理订单超时取消
     * 幂等控制：基于订单ID防重复消费
     */
    @RocketMQMessageListener(
            topic = "${order.timeout.topic:order_timeout_topic}",
            consumerGroup = "${order.timeout.group:order_timeout_consumer_group}",
            delayLevel = "16" // RocketMQ 延迟级别16 = 15分钟
    )
    public void onTimeoutMessage(OrderTimeoutMessage message) {
        long orderId = message.getOrderId();
        log.info("收到订单超时延迟消息, orderId={}", orderId);

        try {
            cancelTimeoutOrder(orderId, "DELAY_MESSAGE");
        } catch (Exception e) {
            log.error("延迟消息处理超时取消异常, orderId={}", orderId, e);
            // 延迟消息消费失败会自动重试，达到最大重试次数后进入死信队列
            throw e;
        }
    }

    /**
     * 发送延迟消息 - 订单创建时调用
     */
    public void sendTimeoutMessage(long orderId) {
        OrderTimeoutMessage message = new OrderTimeoutMessage();
        message.setOrderId(orderId);
        message.setCreateTime(System.currentTimeMillis());
        try {
            rocketMQTemplate.asyncSend(
                    "order_timeout_topic:" + OrderTimeoutMessage.class.getName(),
                    message, new DefaultMQPushConsumerCallback() {
                        @Override
                        public void onSuccess(SendResult sendResult) {
                            log.info("订单超时延迟消息发送成功, orderId={}", orderId);
                        }

                        @Override
                        public void onException(Throwable e) {
                            log.error("订单超时延迟消息发送失败, orderId={}", orderId, e);
                            // 发送失败由定时扫描兜底
                        }
                    });
        } catch (Exception e) {
            log.error("延迟消息发送异常, orderId={}", orderId, e);
        }
    }

    // ==================== 方式二：定时扫描兜底 ====================

    /**
     * 定时扫描超时未支付订单 - 延迟消息的兜底机制
     * 每5分钟扫描一次，处理15分钟前创建但仍未支付的订单
     */
    @Scheduled(fixedRate = 300000)
    public void scanTimeoutOrders() {
        Date threshold = new Date(System.currentTimeMillis()
                - PAY_TIMEOUT_MINUTES * 60 * 1000L);
        List<OrderEntity> timeoutOrders = orderRepository
                .findByMerchantStatusAndCreateTimeBefore(
                        MerchantStatusEnum.WAIT_ACCEPT.getCode(), threshold);

        log.info("定时扫描超时订单, count={}, threshold={}", timeoutOrders.size(), threshold);

        for (OrderEntity order : timeoutOrders) {
            try {
                // 检查是否有支付记录（防止延迟消息和定时扫描的竞争条件）
                if (paymentService.hasPaymentRecord(order.getOrderId())) {
                    log.info("订单已有支付记录，跳过取消, orderId={}", order.getOrderId());
                    continue;
                }
                cancelTimeoutOrder(order.getOrderId(), "SCAN_TASK");
            } catch (Exception e) {
                log.error("定时扫描取消订单异常, orderId={}", order.getOrderId(), e);
            }
        }
    }

    // ==================== 核心取消逻辑 ====================

    /**
     * 执行超时取消 - 核心方法
     * 幂等控制：Redis 分布式锁 + DB 乐观锁双重保障
     */
    @Transactional(rollbackFor = Exception.class)
    public void cancelTimeoutOrder(long orderId, String triggerSource) {
        // 1. 幂等控制：防止延迟消息和定时扫描重复处理
        String idempotentKey = CANCEL_IDEMPOTENT_PREFIX + orderId;
        Boolean locked = idempotencyService.tryLock(idempotentKey, IDEMPOTENT_EXPIRE_SECONDS);
        if (Boolean.FALSE.equals(locked)) {
            log.info("订单超时取消已处理过, 跳过, orderId={}, trigger={}", orderId, triggerSource);
            return;
        }

        try {
            // 2. 查询订单
            OrderEntity order = orderRepository.findById(orderId);
            if (order == null) {
                log.warn("订单不存在, orderId={}", orderId);
                return;
            }

            // 3. 状态校验：只有待接单（未支付）状态才能超时取消
            if (order.getMerchantStatus() != MerchantStatusEnum.WAIT_ACCEPT.getCode()
                    && order.getMerchantStatus() != MerchantStatusEnum.INIT.getCode()) {
                log.info("订单状态非待支付，跳过取消, orderId={}, merchantStatus={}",
                        orderId, order.getMerchantStatus());
                return;
            }

            // 4. 再次校验是否有支付记录（双重检查）
            if (paymentService.hasPaymentRecord(orderId)) {
                log.info("订单已有支付记录，取消中止, orderId={}", orderId);
                idempotencyService.releaseLock(idempotentKey);
                return;
            }

            log.info("开始执行订单超时取消, orderId={}, trigger={}, userId={}",
                    orderId, triggerSource, order.getUserId());

            // 5. CAS 乐观锁更新订单状态（防止并发支付回调与超时取消的竞争）
            int updatedRows = orderRepository.casUpdateMerchantStatus(
                    orderId,
                    MerchantStatusEnum.WAIT_ACCEPT.getCode(),
                    MerchantStatusEnum.REJECTED.getCode(),
                    order.getVersion()
            );
            if (updatedRows == 0) {
                log.warn("CAS更新失败，订单可能已被其他流程处理, orderId={}", orderId);
                idempotencyService.releaseLock(idempotentKey);
                return;
            }

            // 6. 释放库存
            try {
                inventoryService.releaseInventory(orderId);
                log.info("库存释放成功, orderId={}", orderId);
            } catch (Exception e) {
                log.error("库存释放失败, orderId={}, 将由补偿任务重试", orderId, e);
                // 记录补偿任务，不中断流程
                compensationService.recordInventoryReleaseFailure(orderId);
            }

            // 7. 退还优惠券
            if (order.getCouponId() != null) {
                try {
                    couponService.returnCoupon(orderId);
                    log.info("优惠券退回成功, orderId={}, couponId={}",
                            orderId, order.getCouponId());
                } catch (Exception e) {
                    log.error("优惠券退回失败, orderId={}", orderId, e);
                    compensationService.recordCouponReturnFailure(orderId);
                }
            }

            // 8. 更新物流状态为已取消
            OrderContext context = new OrderContext(order);
            stateMachine.transition(context, StatusDimension.LOGISTICS,
                    LogisticsStatusEnum.CANCELED.getCode());

            // 9. 缓存失效
            orderCacheService.invalidateCache(orderId);

            // 10. 发布订单取消事件
            eventPublisher.publish(new OrderTimeoutCancelEvent(
                    orderId, order.getUserId(), order.getMerchantId(), triggerSource));

            // 11. 通知用户（多通道：APP推送 + 短信）
            notifyUserOrderCanceled(order);

            log.info("订单超时取消完成, orderId={}, trigger={}", orderId, triggerSource);

        } catch (BusinessException e) {
            log.warn("订单超时取消业务异常, orderId={}, error={}", orderId, e.getMessage());
            idempotencyService.releaseLock(idempotentKey);
            throw e;
        } catch (Exception e) {
            log.error("订单超时取消系统异常, orderId={}", orderId, e);
            idempotencyService.releaseLock(idempotentKey);
            throw new SystemException("订单取消失败");
        }
    }

    // ==================== 通知逻辑 ====================

    /**
     * 多通道通知用户订单已取消
     */
    private void notifyUserOrderCanceled(OrderEntity order) {
        long orderId = order.getOrderId();
        long userId = order.getUserId();

        // 1. APP推送
        try {
            PushMessage pushMsg = new PushMessage();
            pushMsg.setUserId(userId);
            pushMsg.setTitle("订单已取消");
            pushMsg.setContent(String.format(
                    "您的订单（%d）因超时未支付已自动取消，库存和优惠券已退回", orderId));
            pushMsg.setOrderId(orderId);
            pushService.sendPush(pushMsg);
            log.info("订单取消推送成功, orderId={}, userId={}", orderId, userId);
        } catch (Exception e) {
            log.warn("订单取消推送失败, orderId={}, userId={}", orderId, userId, e);
            // 推送失败不影响主流程
        }

        // 2. 短信通知（推送失败时降级）

            }
        } catch (Exception e) {
            log.warn("订单取消短信通知失败, orderId={}", orderId, e);
        }
    }

/**
 * 库存服务 - 预扣与释放
 */
@Service
@Slf4j
class InventoryService {

    @Autowired
    private RedisTemplate<String, String> redisTemplate;
    @Autowired
    private InventoryRepository inventoryRepository;

    private static final String STOCK_PREFIX = "inventory:stock:";
    private static final String LOCK_PREFIX = "inventory:lock:";

    /**
     * 锁定库存（预扣）
     * Redis 原子扣减 + DB 记录锁定日志
     */
    public InventoryLockResult lockInventory(long merchantId,
                                              List<InventoryLockItem> items,
                                              long orderId) {
        InventoryLockResult result = new InventoryLockResult();
        List<String> lockedKeys = new ArrayList<>();

        try {
            for (InventoryLockItem item : items) {
                String stockKey = STOCK_PREFIX + item.getSkuId();
                Long remaining = redisTemplate.opsForValue().increment(
                        stockKey, -item.getQuantity());
                if (remaining != null && remaining >= 0) {
                    // 记录锁定日志
                    String lockKey = LOCK_PREFIX + orderId + ":" + item.getSkuId();
                    redisTemplate.opsForValue().set(lockKey,
                            String.valueOf(item.getQuantity()), 86400, TimeUnit.SECONDS);
                    lockedKeys.add(lockKey);
                    log.info("库存预扣成功, skuId={}, quantity={}, orderId={}",
                            item.getSkuId(), item.getQuantity(), orderId);
                } else {
                    // 库存不足，回滚已锁定的库存
                    redisTemplate.opsForValue().increment(stockKey, item.getQuantity());
                    rollbackLockedStock(lockedKeys);
                    result.setAllSuccess(false);
                    result.addFailedSku(item.getSkuId());
                    log.warn("库存不足, skuId={}, orderId={}", item.getSkuId(), orderId);
                    return result;
                }
            }
            result.setAllSuccess(true);
            return result;
        } catch (Exception e) {
            log.error("库存锁定异常, orderId={}", orderId, e);
            rollbackLockedStock(lockedKeys);
            result.setAllSuccess(false);
            return result;
        }
    }

    /**
     * 释放库存
     */
    public void releaseInventory(long orderId) {
        Set<String> lockKeys = redisTemplate.keys(LOCK_PREFIX + orderId + ":*");
        if (lockKeys == null || lockKeys.isEmpty()) {
            log.warn("未找到库存锁定记录, orderId={}", orderId);
            return;
        }
        for (String lockKey : lockKeys) {
            String quantityStr = redisTemplate.opsForValue().get(lockKey);
            if (quantityStr == null) continue;
            int quantity = Integer.parseInt(quantityStr);
            String skuId = lockKey.substring(lockKey.lastIndexOf(":") + 1);
            String stockKey = STOCK_PREFIX + skuId;
            redisTemplate.opsForValue().increment(stockKey, quantity);
            redisTemplate.delete(lockKey);
            log.info("库存释放成功, skuId={}, quantity={}, orderId={}",
                    skuId, quantity, orderId);
        }
    }

    /**
     * 解锁库存（订单创建失败时回滚）
     */
    public void unlockInventory(long orderId) {
        releaseInventory(orderId);
    }

    private void rollbackLockedStock(List<String> lockedKeys) {
        for (String lockKey : lockedKeys) {
            String quantityStr = redisTemplate.opsForValue().get(lockKey);
            if (quantityStr == null) continue;
            int quantity = Integer.parseInt(quantityStr);
            String skuId = lockKey.substring(lockKey.lastIndexOf(":") + 1);
            redisTemplate.opsForValue().increment(STOCK_PREFIX + skuId, quantity);
            redisTemplate.delete(lockKey);
        }
    }
}

    @Autowired
    private CompensationLogRepository compensationLogRepo;
    @Autowired
    private InventoryService inventoryService;
    @Autowired
    private CouponService couponService;

    private static final int MAX_RETRY = 5;

    /**
     * 每2分钟扫描补偿任务
     */
    @Scheduled(fixedRate = 120000)
    public void scanCompensationTasks() {
        List<CompensationLog> pendingLogs = compensationLogRepo
                .findByTypeAndRetryCountLessThan("TIMEOUT_CANCEL", MAX_RETRY);

        for (CompensationLog logEntry : pendingLogs) {
            long orderId = logEntry.getOrderId();
            try {
                if (logEntry.getSubType().equals("INVENTORY_RELEASE")) {
                    inventoryService.releaseInventory(orderId);
                    log.info("补偿：库存释放成功, orderId={}", orderId);
                } else if (logEntry.getSubType().equals("COUPON_RETURN")) {
                    couponService.returnCoupon(orderId);
                    log.info("补偿：优惠券退回成功, orderId={}", orderId);
                }
                logEntry.setPhase("COMPENSATED");
                compensationLogRepo.save(logEntry);
            } catch (Exception e) {
                logEntry.setRetryCount(logEntry.getRetryCount() + 1);
                compensationLogRepo.save(logEntry);
                log.error("补偿任务执行失败, orderId={}, retry={}",
                        orderId, logEntry.getRetryCount(), e);
                if (logEntry.getRetryCount() >= MAX_RETRY) {
                    log.error("补偿任务达到最大重试次数, orderId={}, 需人工介入", orderId);
                    alertService.sendAlert(AlertLevel.CRITICAL,
                            "订单超时取消补偿失败，需人工介入: orderId=" + orderId);
                }
            }
        }
    }
```

### 7.3 案例三：订单查询与列表全链路

#### 7.3.1 链路概述

订单查询是用户最高频的操作之一，需要支持多条件查询、分库分表路由、缓存策略、订单详情聚合等。本案例展示从列表查询到详情聚合的完整链路，包含缓存穿透/击穿防护和分库分表路由。

```
多条件查询 -> 分库分表路由 -> 缓存策略 -> 订单详情聚合
    |              |              |              |
    |  状态/时间/  |  userId路由  |  多级缓存    |  并行查询    |
    |  商家筛选    |  orderId路由 |  防穿透/击穿  |  超时降级    |
```

#### 7.3.2 完整代码实现

```java
/**
 * 订单查询与列表全链路服务
 * 涵盖：多条件查询 -> 分库分表路由 -> 缓存策略 -> 订单详情聚合
 */
@Service
@Slf4j
public class OrderQueryFullChainService {

    @Autowired
    private OrderRepository orderRepository;
    @Autowired
    private OrderCacheService orderCacheService;
    @Autowired
    private OrderDatabaseRouter databaseRouter;
    @Autowired
    private OrderShardingStrategy shardingStrategy;
    @Autowired
    private MerchantService merchantService;
    @Autowired
    private DeliveryService deliveryService;
    @Autowired
    private RedisTemplate<String, String> redisTemplate;
    @Autowired
    private ExecutorService orderQueryExecutor;

    private static final String ORDER_LIST_CACHE_PREFIX = "order:list:";
    private static final String ORDER_DETAIL_CACHE_PREFIX = "order:detail:";
    private static final String NULL_CACHE_VALUE = "NULL_CACHE";
    private static final int LIST_CACHE_EXPIRE_SECONDS = 60;
    private static final int DETAIL_CACHE_EXPIRE_SECONDS = 300;
    private static final int NULL_CACHE_EXPIRE_SECONDS = 30;

    // ==================== 第一部分：订单列表查询 ====================

    /**
     * 多条件查询订单列表
     * 支持：用户ID、商家ID、状态、时间范围、分页
     * 缓存策略：列表结果短时缓存 + 防穿透
     */
    public PageResult<OrderListDTO> queryOrderList(OrderQueryRequest request) {
        long userId = request.getUserId();
        int page = request.getPage();
        int size = request.getSize();

        log.info("查询订单列表, userId={}, page={}, size={}, status={}",
                userId, page, size, request.getStatusCode());

        // 1. 缓存Key构建
        String cacheKey = buildListCacheKey(request);

        try {
            // 2. 查缓存
            PageResult<OrderListDTO> cached = tryGetListFromCache(cacheKey);
            if (cached != null) {
                log.info("订单列表命中缓存, userId={}, page={}", userId, page);
                return cached;
            }

            // 3. 分库分表路由
            int dbIndex = shardingStrategy.getDatabaseIndex(userId);
            databaseRouter.route(userId, 0); // 列表查询路由到对应分库

            // 4. 构建查询条件
            OrderQueryCondition condition = buildQueryCondition(request);

            // 5. 查询数据库
            List<OrderEntity> orders;
            long total;
            try {
                orders = orderRepository.findByCondition(condition, page, size);
                total = orderRepository.countByCondition(condition);
            } finally {
                databaseRouter.clear();
            }

            if (orders.isEmpty()) {
                // 防穿透：缓存空结果
                cacheNullResult(cacheKey);
                return PageResult.empty();
            }

            // 6. 聚合关联数据
            OrderListDTO listDTO = aggregateListData(orders);

            // 7. 构建分页结果
            PageResult<OrderListDTO> result = new PageResult<>(
                    Collections.singletonList(listDTO), total);

            // 8. 写入缓存
            cacheListResult(cacheKey, result);

            log.info("查询订单列表完成, userId={}, count={}, total={}",
                    userId, orders.size(), total);
            return result;

        } catch (Exception e) {
            log.error("查询订单列表异常, userId={}", userId, e);
            // 降级：返回空列表
            return PageResult.empty();
        }
    }

    // ==================== 第二部分：订单详情查询 ====================

    /**
     * 查询订单详情
     * 缓存策略：多级缓存（Redis -> DB）+ 防穿透 + 防击穿
     */
    public OrderDetailDTO queryOrderDetail(long orderId, long userId) {
        log.info("查询订单详情, orderId={}, userId={}", orderId, userId);

        // 1. 幂等性控制：防止短时间内的重复查询打爆缓存
        String detailCacheKey = ORDER_DETAIL_CACHE_PREFIX + orderId;

        try {
            // 2. 查缓存
            OrderDetailDTO cached = tryGetDetailFromCache(detailCacheKey);
            if (cached != null) {
                log.info("订单详情命中缓存, orderId={}", orderId);
                return cached;
            }

            // 3. 防击穿：获取分布式锁，只允许一个请求查DB
            String lockKey = "order:detail:lock:" + orderId;
            Boolean locked = redisTemplate.opsForValue()
                    .setIfAbsent(lockKey, "1", 10, TimeUnit.SECONDS);

            if (Boolean.FALSE.equals(locked)) {
                // 未获取到锁，等待后重试读缓存
                Thread.sleep(50);
                cached = tryGetDetailFromCache(detailCacheKey);
                if (cached != null) {
                    return cached;
                }
                // 仍然没有，继续查DB（降级策略）
                log.warn("防击穿锁竞争失败，直接查DB, orderId={}", orderId);
            }

            try {
                // 4. 分库分表路由
                databaseRouter.route(userId, orderId);

                // 5. 查询订单主表
                OrderEntity order = orderRepository.findById(orderId);
                if (order == null) {
                    // 防穿透：缓存空值
                    redisTemplate.opsForValue().set(detailCacheKey, NULL_CACHE_VALUE,
                            NULL_CACHE_EXPIRE_SECONDS, TimeUnit.SECONDS);
                    log.warn("订单不存在, orderId={}", orderId);
                    throw new BusinessException("订单不存在");
                }

                // 6. 权限校验
                if (order.getUserId() != userId) {
                    log.warn("订单查询权限校验失败, orderId={}, userId={}", orderId, userId);
                    throw new BusinessException("无权查看该订单");
                }

                // 7. 并行聚合关联数据
                OrderDetailDTO detailDTO = aggregateDetailData(order);

                // 8. 写入缓存
                cacheDetailResult(detailCacheKey, detailDTO);

                log.info("查询订单详情完成, orderId={}", orderId);
                return detailDTO;

            } finally {
                databaseRouter.clear();
                redisTemplate.delete(lockKey);
            }

        } catch (BusinessException e) {
            throw e;
        } catch (Exception e) {
            log.error("查询订单详情异常, orderId={}", orderId, e);
            throw new SystemException("查询订单详情失败");
        }
    }

    // ==================== 第三部分：数据聚合 ====================

    /**
     * 聚合列表数据 - 并行查询商家、商品、配送信息
     */
    private OrderListDTO aggregateListData(List<OrderEntity> orders) {
        List<Long> orderIds = orders.stream()
                .map(OrderEntity::getOrderId)
                .collect(Collectors.toList());
        List<Long> merchantIds = orders.stream()
                .map(OrderEntity::getMerchantId)
                .distinct()
                .collect(Collectors.toList());

        // 并行查询
        CompletableFuture<Map<Long, MerchantInfo>> merchantsFuture = CompletableFuture
                .supplyAsync(() -> merchantService.batchGet(merchantIds), orderQueryExecutor);

        CompletableFuture<Map<Long, List<OrderItemEntity>>> itemsFuture = CompletableFuture
                .supplyAsync(() -> orderRepository.batchGetItems(orderIds), orderQueryExecutor);

        try {
            CompletableFuture.allOf(merchantsFuture, itemsFuture)
                    .get(3, TimeUnit.SECONDS);

            Map<Long, MerchantInfo> merchants = merchantsFuture.get();
            Map<Long, List<OrderItemEntity>> items = itemsFuture.get();

            // 聚合
            List<OrderListItemDTO> items2 = orders.stream().map(order -> {
                OrderListItemDTO dto = new OrderListItemDTO();
                dto.setOrderId(order.getOrderId());
                dto.setMerchantStatus(order.getMerchantStatus());
                dto.setLogisticsStatus(order.getLogisticsStatus());
                dto.setRefundStatus(order.getRefundStatus());
                dto.setPayAmount(order.getPayAmount());
                dto.setCreateTime(order.getCreateTime());
                dto.setMerchantName(merchants.getOrDefault(
                        order.getMerchantId(), MerchantInfo.unknown()).getName());
                dto.setItems(items.getOrDefault(order.getOrderId(), Collections.emptyList()));
                dto.setDisplayStatus(OrderFlowCode.calculate(
                        order.getMerchantStatus(),
                        order.getLogisticsStatus(),
                        order.getRefundStatus()).getDesc());
                return dto;
            }).collect(Collectors.toList());

            OrderListDTO listDTO = new OrderListDTO();
            listDTO.setOrders(items2);
            return listDTO;

        } catch (TimeoutException e) {
            log.warn("列表数据聚合超时，返回降级数据, orderIds={}", orderIds);
            // 降级：只返回订单基础信息
            return degradeListData(orders);
        } catch (Exception e) {
            log.error("列表数据聚合异常, orderIds={}", orderIds, e);
            return degradeListData(orders);
        }
    }

    /**
     * 聚合详情数据 - 并行查询所有关联信息
     */
    private OrderDetailDTO aggregateDetailData(OrderEntity order) {
        long orderId = order.getOrderId();

        // 并行查询：商家信息、订单商品、配送信息、支付信息、时间线
        CompletableFuture<MerchantInfo> merchantFuture = CompletableFuture
                .supplyAsync(() -> merchantService.getById(order.getMerchantId()), orderQueryExecutor);

        CompletableFuture<List<OrderItemEntity>> itemsFuture = CompletableFuture
                .supplyAsync(() -> orderRepository.findItemsByOrderId(orderId), orderQueryExecutor);

        CompletableFuture<DeliveryInfo> deliveryFuture = CompletableFuture
                .supplyAsync(() -> deliveryService.getInfo(orderId), orderQueryExecutor);

        CompletableFuture<PaymentInfo> paymentFuture = CompletableFuture
                .supplyAsync(() -> orderRepository.findPaymentByOrderId(orderId), orderQueryExecutor);

        CompletableFuture<OrderTimeline> timelineFuture = CompletableFuture
                .supplyAsync(() -> orderRepository.findTimelineByOrderId(orderId), orderQueryExecutor);

        try {
            // 等待所有查询完成，超时5秒
            CompletableFuture.allOf(merchantFuture, itemsFuture, deliveryFuture,
                    paymentFuture, timelineFuture).get(5, TimeUnit.SECONDS);

            OrderDetailDTO dto = new OrderDetailDTO();
            dto.setOrder(order);
            dto.setMerchant(merchantFuture.get());
            dto.setItems(itemsFuture.get());
            dto.setDelivery(deliveryFuture.get());
            dto.setPayment(paymentFuture.get());
            dto.setTimeline(timelineFuture.get());
            dto.setDisplayStatus(OrderFlowCode.calculate(
                    order.getMerchantStatus(),
                    order.getLogisticsStatus(),
                    order.getRefundStatus()));

            return dto;

        } catch (TimeoutException e) {
            log.warn("详情数据聚合超时，返回部分数据, orderId={}", orderId);
            return degradeDetailData(order, merchantFuture, itemsFuture);
        } catch (Exception e) {
            log.error("详情数据聚合异常, orderId={}", orderId, e);
            return degradeDetailData(order, merchantFuture, itemsFuture);
        }
    }

    // ==================== 缓存策略 ====================

    /**
     * 尝试从缓存读取列表
     */
    private PageResult<OrderListDTO> tryGetListFromCache(String cacheKey) {
        String cached = redisTemplate.opsForValue().get(cacheKey);
        if (cached == null) {
            return null;
        }
        if (NULL_CACHE_VALUE.equals(cached)) {
            return PageResult.empty();
        }
        return JSON.parseObject(cached, new TypeReference<PageResult<OrderListDTO>>() {});
    }

    /**
     * 尝试从缓存读取详情
     */
    private OrderDetailDTO tryGetDetailFromCache(String cacheKey) {
        String cached = redisTemplate.opsForValue().get(cacheKey);
        if (cached == null) {
            return null;
        }
        if (NULL_CACHE_VALUE.equals(cached)) {
            return null; // 空缓存不返回详情，让上层处理
        }
        return JSON.parseObject(cached, OrderDetailDTO.class);
    }

    /**
     * 缓存列表结果
     */
    private void cacheListResult(String cacheKey, PageResult<OrderListDTO> result) {
        try {
            redisTemplate.opsForValue().set(cacheKey,
                    JSON.toJSONString(result),
                    LIST_CACHE_EXPIRE_SECONDS, TimeUnit.SECONDS);
        } catch (Exception e) {
            log.warn("缓存列表结果失败, cacheKey={}", cacheKey, e);
        }
    }

    /**
     * 缓存详情结果
     */
    private void cacheDetailResult(String cacheKey, OrderDetailDTO dto) {
        try {
            redisTemplate.opsForValue().set(cacheKey,
                    JSON.toJSONString(dto),
                    DETAIL_CACHE_EXPIRE_SECONDS, TimeUnit.SECONDS);
        } catch (Exception e) {
            log.warn("缓存详情结果失败, cacheKey={}", cacheKey, e);
        }
    }

    /**
     * 缓存空结果（防穿透）
     */
    private void cacheNullResult(String cacheKey) {
        try {
            redisTemplate.opsForValue().set(cacheKey, NULL_CACHE_VALUE,
                    NULL_CACHE_EXPIRE_SECONDS, TimeUnit.SECONDS);
        } catch (Exception e) {
            log.warn("缓存空结果失败, cacheKey={}", cacheKey, e);
        }
    }

    // ==================== 降级策略 ====================

    /**
     * 列表降级：只返回基础信息
     */
    private OrderListDTO degradeListData(List<OrderEntity> orders) {
        List<OrderListItemDTO> items = orders.stream().map(order -> {
            OrderListItemDTO dto = new OrderListItemDTO();
            dto.setOrderId(order.getOrderId());
            dto.setMerchantStatus(order.getMerchantStatus());
            dto.setPayAmount(order.getPayAmount());
            dto.setCreateTime(order.getCreateTime());
            dto.setDisplayStatus(OrderFlowCode.calculate(
                    order.getMerchantStatus(),
                    order.getLogisticsStatus(),
                    order.getRefundStatus()).getDesc());
            dto.setItems(Collections.emptyList());
            dto.setMerchantName("商家信息加载中");
            return dto;
        }).collect(Collectors.toList());

        OrderListDTO listDTO = new OrderListDTO();
        listDTO.setOrders(items);
        listDTO.setDegraded(true);
        return listDTO;
    }

    /**
     * 详情降级：返回已完成查询的部分数据
     */
    private OrderDetailDTO degradeDetailData(OrderEntity order,
                                              CompletableFuture<MerchantInfo> merchantFuture,
                                              CompletableFuture<List<OrderItemEntity>> itemsFuture) {
        OrderDetailDTO dto = new OrderDetailDTO();
        dto.setOrder(order);
        dto.setDegraded(true);

        try {
            if (merchantFuture.isDone()) {
                dto.setMerchant(merchantFuture.get());
            }
        } catch (Exception ignored) {}

        try {
            if (itemsFuture.isDone()) {
                dto.setItems(itemsFuture.get());
            } else {
                dto.setItems(Collections.emptyList());
            }
        } catch (Exception ignored) {}

        dto.setDisplayStatus(OrderFlowCode.calculate(
                order.getMerchantStatus(),
                order.getLogisticsStatus(),
                order.getRefundStatus()));
        return dto;
    }

    // ==================== 辅助方法 ====================

    /**
     * 构建列表缓存Key
     */
    private String buildListCacheKey(OrderQueryRequest request) {
        return ORDER_LIST_CACHE_PREFIX + request.getUserId()
                + ":" + request.getStatusCode()
                + ":" + request.getPage()
                + ":" + request.getSize();
    }

    /**
     * 构建查询条件
     */
    private OrderQueryCondition buildQueryCondition(OrderQueryRequest request) {
        OrderQueryCondition condition = new OrderQueryCondition();
        condition.setUserId(request.getUserId());
        if (request.getMerchantId() != null) {
            condition.setMerchantId(request.getMerchantId());
        }
        if (request.getStatusCode() != null) {
            // 支持新旧状态码
            if (request.getStatusCode() < 100) {
                condition.setLegacyStatusCode(request.getStatusCode());
            } else {
                condition.setFlowCode(request.getStatusCode());
            }
        }
        if (request.getStartTime() != null) {
            condition.setStartTime(request.getStartTime());
        }
        if (request.getEndTime() != null) {
            condition.setEndTime(request.getEndTime());
        }
        return condition;
    }

/**

@Repository
@Slf4j
class OrderShardingRepository {

    @Autowired
    private JdbcTemplate jdbcTemplate;
    @Autowired
    private OrderShardingStrategy shardingStrategy;

    private static final int DATABASE_COUNT = 8;
    private static final int TABLE_COUNT = 16;

    /**
     * 根据用户ID和订单ID路由到对应分库分表查询
     */
    public OrderEntity findById(long orderId, long userId) {
        int dbIndex = shardingStrategy.getDatabaseIndex(userId);
        int tableIndex = shardingStrategy.getTableIndex(orderId);
        String tableName = String.format("t_order_%02d", tableIndex);
        String dataSource = String.format("ds_%d", dbIndex);

        String sql = String.format(
                "SELECT * FROM %s WHERE order_id = ?", tableName);

        log.debug("分库分表查询, dataSource={}, table={}, orderId={}",
                dataSource, tableName, orderId);

        try {
            // 使用动态数据源路由
            DynamicDataSourceContext.setDataSource(dataSource);
            List<OrderEntity> results = jdbcTemplate.query(
                    sql, new Object[]{orderId}, new OrderRowMapper());
            return results.isEmpty() ? null : results.get(0);
        } finally {
            DynamicDataSourceContext.clearDataSource();
        }
    }

    /**
     * 根据用户ID查询订单列表（跨分表）
     * 由于按userId分库，同一用户的订单在同一分库内
     * 但可能分布在不同的分表中，需要跨表查询
     */
    public List<OrderEntity> findByUserId(long userId, int page, int size) {
        int dbIndex = shardingStrategy.getDatabaseIndex(userId);
        String dataSource = String.format("ds_%d", dbIndex);

        // 跨所有分表 UNION 查询
        StringBuilder unionSql = new StringBuilder();
        for (int i = 0; i < TABLE_COUNT; i++) {
            if (i > 0) {
                unionSql.append(" UNION ALL ");
            }
            unionSql.append(String.format(
                    "SELECT * FROM t_order_%02d WHERE user_id = ?", i));
        }
        unionSql.append(" ORDER BY create_time DESC LIMIT ? OFFSET ?");

        log.debug("跨分表查询用户订单, dataSource={}, userId={}, page={}, size={}",
                dataSource, userId, page, size);

        try {
            DynamicDataSourceContext.setDataSource(dataSource);
            Object[] params = new Object[TABLE_COUNT + 2];
            Arrays.fill(params, 0, TABLE_COUNT, userId);
            params[TABLE_COUNT] = size;
            params[TABLE_COUNT + 1] = page * size;
            return jdbcTemplate.query(unionSql.toString(), params, new OrderRowMapper());
        } finally {
            DynamicDataSourceContext.clearDataSource();
        }
    }

    /**
     * CAS 乐观锁更新状态
     */
    public int casUpdateMerchantStatus(long orderId, int expectedStatus,
                                        int newStatus, int version) {
        long userId = orderRepository.findUserIdByOrderId(orderId);
        int dbIndex = shardingStrategy.getDatabaseIndex(userId);
        int tableIndex = shardingStrategy.getTableIndex(orderId);
        String tableName = String.format("t_order_%02d", tableIndex);
        String dataSource = String.format("ds_%d", dbIndex);

        String sql = String.format(
                "UPDATE %s SET merchant_status = ?, update_time = ?, version = version + 1 "
                        + "WHERE order_id = ? AND merchant_status = ? AND version = ?",
                tableName);

        try {
            DynamicDataSourceContext.setDataSource(dataSource);
            return jdbcTemplate.update(sql,
                    newStatus, new Date(), orderId, expectedStatus, version);
        } finally {
            DynamicDataSourceContext.clearDataSource();
        }
    }
}

/**
 * 动态数据源上下文
 */
class DynamicDataSourceContext {
    private static final ThreadLocal<String> CONTEXT = new ThreadLocal<>();

    public static void setDataSource(String dataSource) {
        CONTEXT.set(dataSource);
    }

    public static String getDataSource() {
        return CONTEXT.get();
    }

    public static void clearDataSource() {
        CONTEXT.remove();
    }
}
```
