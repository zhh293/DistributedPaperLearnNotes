# 售后退款系统架构设计

## 一、问题背景

### 1.1 业务场景

售后退款系统处理用户下单后的逆向流程：退货退款、仅退款、换货。一个完整的售后流程：用户发起售后申请 → 商家审核 → 用户退货物流 → 商家验货 → 退款到用户账户。

与正向交易不同，售后是逆向流程，涉及资金退回、库存恢复、优惠权益回收等多个补偿操作，一致性保障是核心难点。

### 1.2 核心挑战

| 挑战维度 | 具体问题 |
|---------|---------|
| 资金安全 | 退款金额不能多退、不能重复退 |
| 补偿一致性 | 退款+恢复库存+回收优惠券+退积分，全成功或全回滚 |
| 状态机复杂 | 售后单状态多（申请→审核→退货→验货→退款→完成） |
| 部分退款 | 多商品订单退一部分，优惠如何分摊 |
| 并发安全 | 同一订单不能同时发起两个售后 |
| 超时处理 | 商家超时不审核自动同意，用户超时不退货自动关闭 |

### 1.3 售后类型

| 类型 | 说明 | 资金 | 库存 | 优惠 |
|------|------|------|------|------|
| 仅退款 | 不退货，直接退款 | 退款 | 不恢复 | 回收 |
| 退货退款 | 退货后退款 | 退款 | 恢复 | 回收 |
| 换货 | 换一个同价商品 | 不退款 | 不恢复 | 不变 |

### 2.2 售后单状态机

```
APPLIED(已申请) ──商家同意──> MERCHANT_AGREED(商家同意)
     │                            │
  商家拒绝                    用户寄回退货
     │                            │
     ▼                            ▼
REJECTED(已拒绝)           GOODS_RETURNED(已退货)
                                  │
                             商家验货通过
                                  │
                                  ▼
                          VERIFIED(已验货)
                                  │
                             执行退款
                                  │
                                  ▼
                           REFUNDING(退款中)
                                  │
                             退款成功
                                  │
                                  ▼
                           COMPLETED(已完成)

超时分支：
  APPLIED → 商家24h未审核 → 自动同意 → MERCHANT_AGREED
  MERCHANT_AGREED → 用户7天未退货 → 自动关闭 → CLOSED
  REFUNDING → 退款失败 → 自动重试
```

## 三、核心链路设计

### 3.1 售后申请

```java
/**
 * 售后申请服务
 *
 * 完整链路：
 * 1. 校验订单状态（已签收才能售后）
 * 2. 校验是否已有进行中的售后单（防止重复申请）
 * 3. 计算退款金额（含优惠分摊）
 * 4. 创建售后单
 * 5. 注册超时任务（商家24h不审核自动同意）
 */
@Service
public class AfterSaleApplyService {

    @Autowired
    private AfterSaleOrderMapper afterSaleMapper;

    @Autowired
    private OrderFeignClient orderClient;

    @Autowired
    private AfterSaleLogMapper logMapper;

    @Transactional(rollbackFor = Exception.class)
    public Long apply(ApplyRequest request, Long userId) {
        // 1. 校验订单
        OrderVO order = orderClient.getOrder(request.getOrderId());
        if (order.getStatus() != OrderStatus.SIGNED.getCode()) {
            throw new BizException("订单未签收，不能申请售后");
        }

        // 2. 校验是否有进行中的售后单
        int activeCount = afterSaleMapper.countActiveByOrder(request.getOrderId());
        if (activeCount > 0) {
            throw new BizException("该订单已有进行中的售后申请");
        }

        // 3. 计算退款金额（优惠分摊）
        RefundCalculation calc = calculateRefund(order, request.getSkuList());

        // 4. 创建售后单
        long afterSaleId = idGenerator.nextId();
        AfterSaleOrder afterSale = new AfterSaleOrder();
        afterSale.setAfterSaleId(afterSaleId);
        afterSale.setOrderId(request.getOrderId());
        afterSale.setUserId(userId);
        afterSale.setType(request.getType());
        afterSale.setStatus(AfterSaleStatus.APPLIED.getCode());
        afterSale.setReason(request.getReason());
        afterSale.setRefundAmount(calc.getRefundAmount());
        afterSale.setSkuList(JSON.toJSONString(request.getSkuList()));
        afterSale.setApplyTime(new Date());
        afterSaleMapper.insert(afterSale);

        // 5. 记录操作流水
        AfterSaleLog log = new AfterSaleLog();
        log.setAfterSaleId(afterSaleId);
        log.setOperator("USER");
        log.setAction("APPLY");
        log.setToStatus(AfterSaleStatus.APPLIED.getCode());
        log.setCreateTime(new Date());
        logMapper.insert(log);

        // 6. 注册超时任务（商家24h不审核自动同意）
        timeoutService.registerMerchantTimeout(afterSaleId);

        return afterSaleId;
    }

    /**
     * 计算退款金额
     *
     * 退款金额 = 商品实付金额 + 分摊的优惠金额
     *
     * 示例：订单100元，优惠券减20元，实付80元
     * 退商品A（原价60元）：
     *   分摊优惠 = 20 × 60/100 = 12元
     *   实付 = 60 - 12 = 48元
     *   退款 = 48元
     */
    private RefundCalculation calculateRefund(OrderVO order,
                                               List<RefundSku> refundSkus) {
        int orderOriginalAmount = order.getOriginalAmount();
        int orderDiscountAmount = order.getDiscountAmount();
        int orderPaidAmount = order.getPaidAmount();

        int totalRefund = 0;

        for (RefundSku refundSku : refundSkus) {
            OrderItem item = order.findItem(refundSku.getSkuId());
            int skuOriginal = item.getPrice() * refundSku.getQuantity();

            // 分摊优惠
            int skuDiscount = (int) ((long) orderDiscountAmount * skuOriginal
                / orderOriginalAmount);
            int skuPaid = skuOriginal - skuDiscount;
            totalRefund += skuPaid;
        }

        // 如果全部退完，退全额（含运费）
        if (isFullRefund(order, refundSkus)) {
            totalRefund = orderPaidAmount;
        }

        RefundCalculation calc = new RefundCalculation();
        calc.setRefundAmount(totalRefund);
        return calc;
    }
}
```

### 3.2 商家审核

```java
/**
 * 商家审核服务
 */
@Service
public class AfterSaleReviewService {

    @Transactional(rollbackFor = Exception.class)
    public void agree(Long afterSaleId, Long merchantId) {
        AfterSaleOrder afterSale = afterSaleMapper.selectByAfterSaleId(afterSaleId);
        validateStatus(afterSale, AfterSaleStatus.APPLIED);

        // 更新状态
        afterSaleMapper.updateStatus(afterSaleId,
            AfterSaleStatus.MERCHANT_AGREED.getCode());
        afterSaleMapper.updateAgreeTime(afterSaleId, new Date());

        // 记录流水
        logService.record(afterSaleId, "MERCHANT", "AGREE",
            AfterSaleStatus.APPLIED.getCode(),
            AfterSaleStatus.MERCHANT_AGREED.getCode());

        // 如果是退货退款，注册用户退货超时任务（7天）
        if (afterSale.getType() == AfterSaleType.RETURN_REFUND.getCode()) {
            timeoutService.registerReturnTimeout(afterSaleId);
        } else if (afterSale.getType() == AfterSaleType.REFUND_ONLY.getCode()) {
            // 仅退款，直接执行退款
            refundService.executeRefund(afterSaleId);
        }
    }

    @Transactional(rollbackFor = Exception.class)
    public void reject(Long afterSaleId, Long merchantId, String reason) {
        AfterSaleOrder afterSale = afterSaleMapper.selectByAfterSaleId(afterSaleId);
        validateStatus(afterSale, AfterSaleStatus.APPLIED);

        afterSaleMapper.updateStatus(afterSaleId,
            AfterSaleStatus.REJECTED.getCode());
        afterSaleMapper.updateRejectReason(afterSaleId, reason);

        logService.record(afterSaleId, "MERCHANT", "REJECT",
            AfterSaleStatus.APPLIED.getCode(),
            AfterSaleStatus.REJECTED.getCode());
    }
}
```

### 3.3 退款执行：补偿事务

**核心难点**：退款涉及多个系统操作（退款+恢复库存+回收优惠券+退积分），必须保证一致性。

**方案**：Saga补偿事务模式——每个操作都有对应的补偿操作，任何一步失败都回滚已执行的步骤。

```java
/**
 * 退款执行服务
 *
 * 补偿事务（Saga）执行顺序：
 * 1. 退款到用户账户（钱包系统）
 * 2. 恢复库存（库存系统）
 * 3. 回收优惠券（优惠券系统）
 * 4. 退回积分（积分系统）
 *
 * 任意一步失败，逆序补偿已执行的步骤
 */
@Service
public class RefundService {

    @Autowired
    private WalletFeignClient walletClient;

    @Autowired
    private InventoryFeignClient inventoryClient;

    @Autowired
    private CouponFeignClient couponClient;

    @Autowired
    private PointFeignClient pointClient;

    @Autowired
    private RefundRecordMapper refundRecordMapper;

    /**
     * 执行退款
     */
    @Transactional(rollbackFor = Exception.class)
    public void executeRefund(Long afterSaleId) {
        AfterSaleOrder afterSale = afterSaleMapper.selectByAfterSaleId(afterSaleId);

        // 状态校验
        if (afterSale.getStatus() != AfterSaleStatus.VERIFIED.getCode()
            && afterSale.getStatus() != AfterSaleStatus.MERCHANT_AGREED.getCode()) {
            throw new BizException("当前状态不允许退款");
        }

        // 更新状态为退款中
        afterSaleMapper.updateStatus(afterSaleId,
            AfterSaleStatus.REFUNDING.getCode());

        // 创建退款记录
        long refundId = idGenerator.nextId();
        RefundRecord record = new RefundRecord();
        record.setRefundId(refundId);
        record.setAfterSaleId(afterSaleId);
        record.setOrderId(afterSale.getOrderId());
        record.setUserId(afterSale.getUserId());
        record.setRefundAmount(afterSale.getRefundAmount());
        record.setRefundType("ORIGINAL");
        record.setStatus(0); // 处理中
        record.setCreateTime(new Date());
        refundRecordMapper.insert(record);

        // Saga补偿事务执行
        boolean refundDone = false;
        boolean inventoryDone = false;
        boolean couponDone = false;
        boolean pointDone = false;

        try {
            // Step 1: 退款
            walletClient.refund(RefundRequest.builder()
                .userId(afterSale.getUserId())
                .amount(afterSale.getRefundAmount())
                .orderId(afterSale.getOrderId())
                .refundId(refundId)
                .build());
            refundDone = true;

            // Step 2: 恢复库存
            if (afterSale.getType() != AfterSaleType.REFUND_ONLY.getCode()) {
                inventoryClient.restore(RestoreRequest.builder()
                    .orderId(afterSale.getOrderId())
                    .skuList(JSON.parseArray(afterSale.getSkuList(), RefundSku.class))
                    .build());
            }
            inventoryDone = true;

            // Step 3: 回收优惠券
            couponClient.reclaim(ReclaimRequest.builder()
                .orderId(afterSale.getOrderId())
                .build());
            couponDone = true;

            // Step 4: 退回积分
            pointClient.restore(RestoreRequest.builder()
                .orderId(afterSale.getOrderId())
                .userId(afterSale.getUserId())
                .build());
            pointDone = true;

            // 全部成功，更新状态
            refundRecordMapper.updateStatus(refundId, 1); // 成功
            afterSaleMapper.updateStatus(afterSaleId,
                AfterSaleStatus.COMPLETED.getCode());
            afterSaleMapper.updateCompleteTime(afterSaleId, new Date());

        } catch (Exception e) {
            log.error("退款执行失败, afterSaleId={}", afterSaleId, e);

            // 逆序补偿
            if (pointDone) {
                try { pointClient.cancelRestore(afterSale.getOrderId()); }
                catch (Exception ex) { log.error("积分补偿失败", ex); }
            }
            if (couponDone) {
                try { couponClient.cancelReclaim(afterSale.getOrderId()); }
                catch (Exception ex) { log.error("优惠券补偿失败", ex); }
            }
            if (inventoryDone) {
                try { inventoryClient.cancelRestore(afterSale.getOrderId()); }
                catch (Exception ex) { log.error("库存补偿失败", ex); }
            }
            if (refundDone) {
                try { walletClient.cancelRefund(refundId); }
                catch (Exception ex) { log.error("退款补偿失败", ex); }
            }

            // 标记退款失败，等待重试
            refundRecordMapper.updateStatus(refundId, 2); // 失败
            throw e;
        }
    }
}
```

### 3.4 超时处理

```java
/**
 * 售后超时处理服务
 *
 * 超时规则：
 * 1. 商家24h不审核 → 自动同意
 * 2. 用户7天不退货 → 自动关闭
 * 3. 退款失败 → 自动重试3次
 */
@Service
public class AfterSaleTimeoutService {

    @Scheduled(fixedRate = 60000) // 每分钟扫描
    public void scanTimeout() {
        // 1. 商家超时未审核（24小时）
        List<AfterSaleOrder> merchantTimeout = afterSaleMapper
            .selectMerchantTimeout(
                DateUtils.addHours(new Date(), -24));
        for (AfterSaleOrder as : merchantTimeout) {
            try {
                autoAgree(as.getAfterSaleId());
            } catch (Exception e) {
                log.error("自动同意失败, id={}", as.getAfterSaleId(), e);
            }
        }

        // 2. 用户超时未退货（7天）
        List<AfterSaleOrder> returnTimeout = afterSaleMapper
            .selectReturnTimeout(
                DateUtils.addDays(new Date(), -7));
        for (AfterSaleOrder as : returnTimeout) {
            try {
                autoClose(as.getAfterSaleId());
            } catch (Exception e) {
                log.error("自动关闭失败, id={}", as.getAfterSaleId(), e);
            }
        }

        // 3. 退款失败重试
        List<RefundRecord> failedRefunds = refundRecordMapper
            .selectFailedWithRetry(3);
        for (RefundRecord record : failedRefunds) {
            try {
                refundService.executeRefund(record.getAfterSaleId());
                refundRecordMapper.incrementRetryCount(record.getRefundId());
            } catch (Exception e) {
                log.error("退款重试失败, refundId={}", record.getRefundId(), e);
                refundRecordMapper.incrementRetryCount(record.getRefundId());
            }
        }
    }

    /**
     * 商家超时自动同意
     */
    @Transactional(rollbackFor = Exception.class)
    public void autoAgree(Long afterSaleId) {
        AfterSaleOrder afterSale = afterSaleMapper.selectByAfterSaleId(afterSaleId);
        if (afterSale.getStatus() != AfterSaleStatus.APPLIED.getCode()) {
            return; // 状态已变更
        }

        afterSaleMapper.updateStatus(afterSaleId,
            AfterSaleStatus.MERCHANT_AGREED.getCode());
        afterSaleMapper.updateAgreeTime(afterSaleId, new Date());

        logService.record(afterSaleId, "SYSTEM", "AUTO_AGREE",
            AfterSaleStatus.APPLIED.getCode(),
            AfterSaleStatus.MERCHANT_AGREED.getCode());

        // 仅退款直接退款
        if (afterSale.getType() == AfterSaleType.REFUND_ONLY.getCode()) {
            refundService.executeRefund(afterSaleId);
        }
    }

    /**
     * 用户超时自动关闭
     */
    @Transactional(rollbackFor = Exception.class)
    public void autoClose(Long afterSaleId) {
        AfterSaleOrder afterSale = afterSaleMapper.selectByAfterSaleId(afterSaleId);
        if (afterSale.getStatus() != AfterSaleStatus.MERCHANT_AGREED.getCode()) {
            return;
        }

        afterSaleMapper.updateStatus(afterSaleId,
            AfterSaleStatus.CLOSED.getCode());

        logService.record(afterSaleId, "SYSTEM", "AUTO_CLOSE",
            AfterSaleStatus.MERCHANT_AGREED.getCode(),
            AfterSaleStatus.CLOSED.getCode());
    }
}
```

### 3.5 退款防重

```java
/**
 * 退款防重
 *
 * 通过退款记录表的唯一索引 + 幂等校验保证不重复退款
 */
@Service
public class RefundIdempotentService {

    /**
     * 检查是否已退款
     */
    public boolean isRefunded(Long afterSaleId) {
        RefundRecord record = refundRecordMapper.selectByAfterSaleId(afterSaleId);
        return record != null && record.getStatus() == 1; // 1=成功
    }

    /**
     * 检查是否正在退款中
     */
    public boolean isRefunding(Long afterSaleId) {
        RefundRecord record = refundRecordMapper.selectByAfterSaleId(afterSaleId);
        return record != null && record.getStatus() == 0; // 0=处理中
    }
}
```

## 四、退款金额分摊计算

```java
/**
 * 退款金额分摊计算
 *
 * 问题：订单100元，含商品A(60元)和商品B(40元)，
 *       优惠券减20元，实付80元
 *       用户退商品A，退款金额是多少？
 *
 * 计算：
 *   A分摊优惠 = 20 × 60/100 = 12元
 *   A实付 = 60 - 12 = 48元
 *   退款 = 48元
 */
@Service
public class RefundAllocationService {

    public RefundAllocation calculate(OrderVO order, List<RefundSku> refundSkus) {
        int orderOriginal = order.getOriginalAmount();
        int orderDiscount = order.getDiscountAmount();
        int orderShipping = order.getShippingFee();

        int totalRefund = 0;
        Map<Long, Integer> skuRefundMap = new HashMap<>();

        for (RefundSku refundSku : refundSkus) {
            OrderItem item = order.findItem(refundSku.getSkuId());
            if (item == null) continue;

            int skuOriginal = item.getPrice() * refundSku.getQuantity();

            // 按比例分摊优惠
            int skuDiscount = (int) ((long) orderDiscount * skuOriginal / orderOriginal);

            // SKU实付 = 原价 - 分摊优惠
            int skuPaid = skuOriginal - skuDiscount;

            skuRefundMap.put(refundSku.getSkuId(), skuPaid);
            totalRefund += skuPaid;
        }

        // 如果全部退完，退运费
        if (isFullRefund(order, refundSkus)) {
            totalRefund += orderShipping;
        }

        RefundAllocation allocation = new RefundAllocation();
        allocation.setTotalRefund(totalRefund);
        allocation.setSkuRefundMap(skuRefundMap);
        allocation.setIncludeShipping(isFullRefund(order, refundSkus));
        return allocation;
    }

    private boolean isFullRefund(OrderVO order, List<RefundSku> refundSkus) {
        Set<Long> refundSkuIds = refundSkus.stream()
            .map(RefundSku::getSkuId).collect(Collectors.toSet());
        return order.getItems().stream()
            .allMatch(item -> refundSkuIds.contains(item.getSkuId()));
    }
}
```

## 五、最佳实践与总结

### 5.1 核心设计决策

| 决策点 | 选择 | 理由 |
|--------|------|------|
| 退款一致性 | Saga补偿事务 | 跨系统操作的一致性保障 |
| 退款防重 | 退款记录表唯一索引 | 绝不重复退款 |
| 优惠分摊 | 按比例分摊 | 退款金额公平 |
| 超时处理 | 定时任务扫描 | 自动化处理 |
| 状态机 | 严格状态流转校验 | 防止非法状态跳转 |
| 退款方式 | 原路退回优先 | 用户体验好 |

### 5.2 容易踩的坑

1. **重复退款**：退款失败重试时没有幂等校验，导致重复退款
2. **补偿事务不完整**：退款成功但库存恢复失败，且没有补偿机制
3. **优惠分摊错误**：退款金额没有按比例扣除优惠，导致退款金额大于实付
4. **状态机不严格**：已完成的售后单还能被修改，导致数据不一致
5. **超时任务遗漏**：商家超时不审核没有自动处理，用户等待时间长
6. **部分退款运费**：部分退款时多退了运费，只有全退才退运费

### 案例一：用户发起仅退款全链路

#### 6.1.1 场景描述

```

→ 支付渠道回调 → 退款到账 → 库存归还/优惠券退回 → 完成
```

#### 6.1.3 订单状态校验与退款窗口期

```java
/**
 * 售后申请服务（全链路实现）
 */
@Service
@Slf4j
public class AfterSaleApplyService {

    @Autowired
    private OrderFeignClient orderClient;

    @Autowired
    private AfterSaleOrderMapper afterSaleMapper;

    @Autowired
    private AfterSaleLogMapper logMapper;

    @Autowired
    private RefundAllocationService allocationService;

    @Autowired
    private AfterSaleTimeoutService timeoutService;

    @Autowired
    private AutoReviewService autoReviewService;

    @Autowired
    private RedissonClient redissonClient;

    /** 售后申请窗口期：签收后15天内 */
    private static final int AFTER_SALE_WINDOW_DAYS = 15;

    /** 分布式锁前缀 */
    private static final String APPLY_LOCK_PREFIX = "lock:aftersale:apply:";

    @Transactional(rollbackFor = Exception.class)
    public Long apply(ApplyRequest request, Long userId) {
        String lockKey = APPLY_LOCK_PREFIX + request.getOrderId();
        RLock lock = redissonClient.getLock(lockKey);

        try {
            // 分布式锁，防止同一订单并发发起售后
            if (!lock.tryLock(3, TimeUnit.SECONDS)) {
                throw new BizException("ORDER_PROCESSING", "订单正在处理中，请稍后重试");
            }

            return doApply(request, userId);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new BizException("SYSTEM_ERROR", "系统繁忙，请稍后重试");
        } finally {
            if (lock.isHeldByCurrentThread()) {
                lock.unlock();
            }
        }
    }

    private Long doApply(ApplyRequest request, Long userId) {
        // ========== Step 1: 校验订单 ==========
        OrderVO order = orderClient.getOrder(request.getOrderId());
        if (order == null) {
            throw new BizException("ORDER_NOT_FOUND", "订单不存在");
        }
        if (!order.getUserId().equals(userId)) {
            throw new BizException("PERMISSION_DENIED", "无权操作此订单");
        }

        // 订单状态校验：只有已签收的订单才能售后
        if (order.getStatus() != OrderStatus.SIGNED.getCode()) {
            throw new BizException("ORDER_STATUS_INVALID",
                "订单状态不允许售后，当前状态：" + order.getStatus());
        }

        // 退款窗口期校验：签收后15天内可申请
        if (order.getSignTime() != null) {
            long daysSinceSign = (System.currentTimeMillis()
                - order.getSignTime().getTime()) / (1000 * 60 * 60 * 24);
            if (daysSinceSign > AFTER_SALE_WINDOW_DAYS) {
                throw new BizException("WINDOW_EXPIRED",
                    "已超过售后申请期限（签收后" + AFTER_SALE_WINDOW_DAYS + "天）");
            }
        }

        // ========== Step 2: 校验是否有进行中的售后单 ==========
        int activeCount = afterSaleMapper.countActiveByOrder(request.getOrderId());
        if (activeCount > 0) {
            throw new BizException("DUPLICATE_APPLY", "该订单已有进行中的售后申请");
        }

        // ========== Step 3: 退款金额校验（优惠分摊计算） ==========
        RefundAllocation allocation = allocationService.calculate(order, request.getSkuList());
        int calculatedAmount = allocation.getTotalRefund();

        // 前端传入的退款金额必须与后端计算一致
        if (request.getRefundAmount() != null
            && !request.getRefundAmount().equals((long) calculatedAmount)) {
            log.warn("退款金额不一致, 前端={}, 后端计算={}, orderId={}",
                request.getRefundAmount(), calculatedAmount, request.getOrderId());
            throw new BizException("AMOUNT_MISMATCH",
                "退款金额与实际不符，请刷新后重试");
        }

        // 退款金额不能超过订单实付金额
        if (calculatedAmount > order.getPaidAmount()) {
            log.error("退款金额超过实付, refund={}, paid={}, orderId={}",
                calculatedAmount, order.getPaidAmount(), request.getOrderId());
            throw new BizException("AMOUNT_EXCEED", "退款金额不能超过实付金额");
        }

        // 退款金额必须大于0
        if (calculatedAmount <= 0) {
            throw new BizException("AMOUNT_INVALID", "退款金额必须大于0");
        }

        // ========== Step 4: 创建售后单 ==========
        long afterSaleId = idGenerator.nextId();
        AfterSaleOrder afterSale = new AfterSaleOrder();
        afterSale.setAfterSaleId(afterSaleId);
        afterSale.setOrderId(request.getOrderId());
        afterSale.setUserId(userId);
        afterSale.setType(request.getType());
        afterSale.setStatus(AfterSaleStatus.APPLIED.getCode());
        afterSale.setReason(request.getReason());
        afterSale.setRefundAmount(calculatedAmount);
        afterSale.setSkuList(JSON.toJSONString(request.getSkuList()));
        afterSale.setEvidenceImages(JSON.toJSONString(request.getEvidenceImages()));
        afterSale.setApplyTime(new Date());
        afterSaleMapper.insert(afterSale);

        // 记录操作流水
        recordLog(afterSaleId, "USER", "APPLY",
            null, AfterSaleStatus.APPLIED.getCode(),
            "用户发起售后申请，退款金额：" + calculatedAmount + "分");

        log.info("售后单创建成功, afterSaleId={}, orderId={}, refundAmount={}",
            afterSaleId, request.getOrderId(), calculatedAmount);

        // ========== Step 5: 提交审核流程 ==========
        // 先走自动审核规则，不通过再转人工
        autoReviewService.tryAutoReview(afterSaleId);

        // ========== Step 6: 注册超时任务 ==========
        // 商家24h不审核自动同意
        timeoutService.registerMerchantTimeout(afterSaleId);

        return afterSaleId;
    }

    private void recordLog(Long afterSaleId, String operator, String action,
                           Integer fromStatus, Integer toStatus, String remark) {
        AfterSaleLog log = new AfterSaleLog();

        log.setAction(action);
        log.setFromStatus(fromStatus);
        log.setToStatus(toStatus);
        log.setRemark(remark);
        log.setCreateTime(new Date());
        logMapper.insert(log);
    }
}
```

#### 6.1.4 自动审核规则与人工审核

```java
/**
 * 自动审核服务
 *
 * 自动审核规则（全部满足才自动通过）：
 * 1. 退款金额 <= 50元（小额免审）
 * 2. 用户历史退款率 < 30%
 * 3. 订单无客诉标记
 * 4. 商品支持7天无理由
 * 5. 凭证图片不为空（质量问题需要凭证）
 *
 * 不满足自动审核条件则转人工审核
 */
@Service
@Slf4j
public class AutoReviewService {

    @Autowired
    private AfterSaleOrderMapper afterSaleMapper;

    @Autowired
    private UserFeignClient userClient;

    @Autowired
    private RiskFeignClient riskClient;

    @Autowired
    private RefundService refundService;

    @Autowired
    private AfterSaleLogMapper logMapper;

    /** 自动审核金额阈值（分） */
    private static final int AUTO_REFUND_THRESHOLD = 5000;

    /** 退款率阈值 */
    private static final double REFUND_RATE_THRESHOLD = 0.30;

    /**
     * 尝试自动审核
     */
    @Transactional(rollbackFor = Exception.class)
    public void tryAutoReview(Long afterSaleId) {
        AfterSaleOrder afterSale = afterSaleMapper.selectByAfterSaleId(afterSaleId);
        if (afterSale == null
            || afterSale.getStatus() != AfterSaleStatus.APPLIED.getCode()) {
            return;
        }

        boolean autoPass = checkAutoReviewRules(afterSale);

        if (autoPass) {
            log.info("自动审核通过, afterSaleId={}", afterSaleId);

            // 更新状态为商家同意（系统自动）
            afterSaleMapper.updateStatus(afterSaleId,
                AfterSaleStatus.MERCHANT_AGREED.getCode());
            afterSaleMapper.updateAgreeTime(afterSaleId, new Date());

            recordLog(afterSaleId, "SYSTEM", "AUTO_AGREE",
                AfterSaleStatus.APPLIED.getCode(),
                AfterSaleStatus.MERCHANT_AGREED.getCode(),
                "系统自动审核通过");

            // 仅退款：直接执行退款
            if (afterSale.getType() == AfterSaleType.REFUND_ONLY.getCode()) {
                refundService.executeRefund(afterSaleId);
            }
        } else {
            log.info("转人工审核, afterSaleId={}", afterSaleId);
            // 等待商家人工审核，不做状态变更
            recordLog(afterSaleId, "SYSTEM", "TRANSFER_MANUAL",
                AfterSaleStatus.APPLIED.getCode(),
                AfterSaleStatus.APPLIED.getCode(),
                "不满足自动审核条件，转人工审核");
        }
    }

    /**
     * 自动审核规则判断
     */
    private boolean checkAutoReviewRules(AfterSaleOrder afterSale) {
        // 规则1: 退款金额 <= 50元
        if (afterSale.getRefundAmount() > AUTO_REFUND_THRESHOLD) {
            log.debug("自动审核不通过-金额超限, afterSaleId={}, amount={}",
                afterSale.getAfterSaleId(), afterSale.getRefundAmount());
            return false;
        }

        // 规则2: 用户历史退款率 < 30%
        UserRiskVO userRisk = userClient.getUserRisk(afterSale.getUserId());
        if (userRisk != null && userRisk.getRefundRate() >= REFUND_RATE_THRESHOLD) {
            log.debug("自动审核不通过-退款率过高, afterSaleId={}, rate={}",
                afterSale.getAfterSaleId(), userRisk.getRefundRate());
            return false;
        }

        // 规则3: 风控检查（是否有客诉、欺诈标记）
        RiskResult risk = riskClient.checkAfterSaleRisk(afterSale.getAfterSaleId(),
            afterSale.getUserId(), afterSale.getOrderId());
        if (risk != null && risk.isBlocked()) {
            log.warn("自动审核不通过-风控拦截, afterSaleId={}, reason={}",
                afterSale.getAfterSaleId(), risk.getReason());
            return false;
        }

        // 规则4: 凭证图片校验（质量问题需要凭证）
        if (StringUtils.isEmpty(afterSale.getEvidenceImages())) {
            log.debug("自动审核不通过-无凭证图片, afterSaleId={}",
                afterSale.getAfterSaleId());
            return false;
        }

        return true;
    }

    private void recordLog(Long afterSaleId, String operator, String action,
                           Integer fromStatus, Integer toStatus, String remark) {
        AfterSaleLog log = new AfterSaleLog();
        log.setAfterSaleId(afterSaleId);
        log.setOperator(operator);
        log.setAction(action);
        log.setFromStatus(fromStatus);
        log.setToStatus(toStatus);
        log.setRemark(remark);
        log.setCreateTime(new Date());
        logMapper.insert(log);
    }
}
```

#### 6.1.5 审核通过后发起退款（原路退回）

```java
/**
 * 退款执行服务（全链路实现）
 *
 * 补偿事务（Saga）执行顺序：
 * 1. 调用支付系统原路退回
 * 2. 等待支付渠道异步回调
 * 3. 回调成功后执行补偿操作（库存归还、优惠券退回、积分退回）
 * 4. 任意一步失败，逆序补偿 + 自动重试
 */
@Service
@Slf4j
public class RefundService {

    @Autowired
    private AfterSaleOrderMapper afterSaleMapper;

    @Autowired
    private RefundRecordMapper refundRecordMapper;

    @Autowired
    private PaymentFeignClient paymentClient;

    @Autowired
    private InventoryFeignClient inventoryClient;

    @Autowired
    private CouponFeignClient couponClient;

    @Autowired
    private PointFeignClient pointClient;

    @Autowired
    private AfterSaleLogMapper logMapper;

    @Autowired
    private IdGenerator idGenerator;

    /** 退款最大重试次数 */
    private static final int MAX_RETRY_COUNT = 3;

    /** 退款幂等锁前缀 */
    private static final String REFUND_LOCK_PREFIX = "lock:refund:";

    /**
     * 执行退款
     */
    public void executeRefund(Long afterSaleId) {
        String lockKey = REFUND_LOCK_PREFIX + afterSaleId;
        RLock lock = redissonClient.getLock(lockKey);

        try {
            if (!lock.tryLock(5, 30, TimeUnit.SECONDS)) {
                log.warn("退款锁获取失败, afterSaleId={}", afterSaleId);
                throw new BizException("REFUND_PROCESSING", "退款正在处理中");
            }

            doExecuteRefund(afterSaleId);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new BizException("SYSTEM_ERROR", "系统繁忙");
        } finally {
            if (lock.isHeldByCurrentThread()) {
                lock.unlock();
            }
        }
    }

    private void doExecuteRefund(Long afterSaleId) {
        AfterSaleOrder afterSale = afterSaleMapper.selectByAfterSaleId(afterSaleId);
        if (afterSale == null) {
            throw new BizException("AFTER_SALE_NOT_FOUND", "售后单不存在");
        }

        // 幂等校验：已成功的退款不重复执行
        RefundRecord existRecord = refundRecordMapper.selectByAfterSaleId(afterSaleId);
        if (existRecord != null && existRecord.getStatus() == 1) {
            log.info("退款已完成，跳过执行, afterSaleId={}", afterSaleId);
            return;
        }

        // 状态校验
        if (afterSale.getStatus() != AfterSaleStatus.MERCHANT_AGREED.getCode()
            && afterSale.getStatus() != AfterSaleStatus.VERIFIED.getCode()) {
            throw new BizException("STATUS_INVALID", "当前状态不允许退款");
        }

        // 更新售后单状态为退款中
        afterSaleMapper.updateStatus(afterSaleId,
            AfterSaleStatus.REFUNDING.getCode());

        // 创建退款记录
        long refundId = idGenerator.nextId();
        RefundRecord record = new RefundRecord();
        record.setRefundId(refundId);
        record.setAfterSaleId(afterSaleId);
        record.setOrderId(afterSale.getOrderId());
        record.setUserId(afterSale.getUserId());
        record.setRefundAmount(afterSale.getRefundAmount());
        record.setRefundType("ORIGINAL"); // 原路退回
        record.setStatus(0); // 处理中
        record.setCreateTime(new Date());
        refundRecordMapper.insert(record);

        recordLog(afterSaleId, "SYSTEM", "REFUND_START",
            AfterSaleStatus.MERCHANT_AGREED.getCode(),
            AfterSaleStatus.REFUNDING.getCode(),
            "发起退款，refundId=" + refundId);

        // 调用支付系统发起原路退回
        try {
            PaymentRefundRequest payRequest = PaymentRefundRequest.builder()
                .refundId(refundId)
                .orderId(afterSale.getOrderId())
                .userId(afterSale.getUserId())
                .refundAmount(afterSale.getRefundAmount())
                .build();

            PaymentRefundResponse response = paymentClient.refund(payRequest);

            if (response.isSuccess()) {
                // 支付渠道已受理，等待异步回调
                refundRecordMapper.updateTradeNo(refundId, response.getTradeNo());
                log.info("退款请求已受理, refundId={}, tradeNo={}",
                    refundId, response.getTradeNo());
            } else {
                // 支付渠道直接拒绝
                log.error("退款请求被拒绝, refundId={}, code={}, msg={}",
                    refundId, response.getErrorCode(), response.getErrorMsg());
                refundRecordMapper.updateStatus(refundId, 2); // 失败
                handleRefundFailure(afterSaleId, refundId, response.getErrorMsg());
            }
        } catch (Exception e) {
            log.error("调用支付系统退款异常, refundId={}", refundId, e);
            refundRecordMapper.updateStatus(refundId, 2); // 失败
            handleRefundFailure(afterSaleId, refundId, e.getMessage());
        }
    }

    /**
     * 退款到账回调处理
     *
     * 支付渠道异步通知退款结果，此方法需保证幂等
     */
    @Transactional(rollbackFor = Exception.class)
    public void handleRefundCallback(RefundCallbackRequest callback) {
        log.info("收到退款回调, refundId={}, status={}, tradeNo={}",
            callback.getRefundId(), callback.getStatus(), callback.getTradeNo());

        // 幂等校验：查询退款记录
        RefundRecord record = refundRecordMapper.selectByRefundId(callback.getRefundId());
        if (record == null) {
            log.error("退款记录不存在, refundId={}", callback.getRefundId());
            throw new BizException("RECORD_NOT_FOUND", "退款记录不存在");
        }

        // 幂等：已处理的回调不重复处理
        if (record.getStatus() != 0) {
            log.info("退款回调已处理，跳过, refundId={}, status={}",
                callback.getRefundId(), record.getStatus());
            return;
        }

        Long afterSaleId = record.getAfterSaleId();

        if (callback.isSuccess()) {
            // ========== 退款成功，执行补偿操作 ==========

            // Step 1: 更新退款记录状态
            refundRecordMapper.updateStatus(record.getRefundId(), 1); // 成功
            refundRecordMapper.updateCompleteTime(record.getRefundId(), new Date());

            // Step 2: 执行补偿操作（库存归还、优惠券退回、积分退回）
            boolean compensateSuccess = executeCompensation(afterSaleId, record);

            if (compensateSuccess) {
                // 全部成功，更新售后单状态为已完成
                afterSaleMapper.updateStatus(afterSaleId,
                    AfterSaleStatus.COMPLETED.getCode());
                afterSaleMapper.updateCompleteTime(afterSaleId, new Date());

                recordLog(afterSaleId, "SYSTEM", "REFUND_COMPLETE",
                    AfterSaleStatus.REFUNDING.getCode(),
                    AfterSaleStatus.COMPLETED.getCode(),
                    "退款及补偿操作全部完成");

                log.info("退款全链路完成, afterSaleId={}, refundId={}",
                    afterSaleId, record.getRefundId());
            } else {
                // 补偿操作部分失败，记录异常，等待补偿任务重试
                log.error("补偿操作部分失败, afterSaleId={}", afterSaleId);
                recordLog(afterSaleId, "SYSTEM", "COMPENSATE_PARTIAL_FAIL",
                    AfterSaleStatus.REFUNDING.getCode(),
                    AfterSaleStatus.REFUNDING.getCode(),
                    "退款已成功但补偿操作部分失败，等待补偿任务重试");
            }
        } else {
            // ========== 退款失败 ==========
            refundRecordMapper.updateStatus(record.getRefundId(), 2); // 失败
            handleRefundFailure(afterSaleId, record.getRefundId(),
                callback.getFailReason());
        }
    }

    /**
     * 执行补偿操作（库存归还、优惠券退回、积分退回）
     *
     * 采用Saga补偿事务模式，任何一步失败逆序回滚
     */
    private boolean executeCompensation(Long afterSaleId, RefundRecord record) {
        AfterSaleOrder afterSale = afterSaleMapper.selectByAfterSaleId(afterSaleId);

        boolean inventoryDone = false;
        boolean couponDone = false;
        boolean pointDone = false;

        try {
            // Step 1: 库存归还（仅退款不归还库存）
            if (afterSale.getType() != AfterSaleType.REFUND_ONLY.getCode()) {
                List<RefundSku> skuList = JSON.parseArray(
                    afterSale.getSkuList(), RefundSku.class);
                inventoryClient.restore(RestoreRequest.builder()
                    .orderId(afterSale.getOrderId())
                    .skuList(skuList)
                    .build());
            }
            inventoryDone = true;
            log.info("库存归还完成, afterSaleId={}", afterSaleId);

            // Step 2: 优惠券退回
            couponClient.reclaim(ReclaimRequest.builder()
                .orderId(afterSale.getOrderId())
                .userId(afterSale.getUserId())
                .build());
            couponDone = true;
            log.info("优惠券退回完成, afterSaleId={}", afterSaleId);

            // Step 3: 积分退回
            pointClient.restore(RestoreRequest.builder()
                .orderId(afterSale.getOrderId())
                .userId(afterSale.getUserId())
                .build());
            pointDone = true;
            log.info("积分退回完成, afterSaleId={}", afterSaleId);

            return true;

        } catch (Exception e) {
            log.error("补偿操作失败, afterSaleId={}, inv={}, coupon={}, point={}",
                afterSaleId, inventoryDone, couponDone, pointDone, e);

            // 逆序补偿
            if (pointDone) {
                try {
                    pointClient.cancelRestore(afterSale.getOrderId());
                } catch (Exception ex) {
                    log.error("积分补偿回滚失败, orderId={}",
                        afterSale.getOrderId(), ex);
                }
            }
            if (couponDone) {
                try {
                    couponClient.cancelReclaim(afterSale.getOrderId());
                } catch (Exception ex) {
                    log.error("优惠券补偿回滚失败, orderId={}",
                        afterSale.getOrderId(), ex);
                }
            }
            if (inventoryDone) {
                try {
                    inventoryClient.cancelRestore(afterSale.getOrderId());
                } catch (Exception ex) {
                    log.error("库存补偿回滚失败, orderId={}",
                        afterSale.getOrderId(), ex);
                }
            }

            return false;
        }
    }

    /**
     * 退款失败处理
     */
    private void handleRefundFailure(Long afterSaleId, Long refundId, String reason) {
        recordLog(afterSaleId, "SYSTEM", "REFUND_FAIL",
            AfterSaleStatus.REFUNDING.getCode(),
            AfterSaleStatus.REFUNDING.getCode(),
            "退款失败：" + reason);

        // 检查重试次数
        RefundRecord record = refundRecordMapper.selectByRefundId(refundId);
        if (record.getRetryCount() < MAX_RETRY_COUNT) {
            // 等待定时任务自动重试
            log.info("退款等待重试, refundId={}, retryCount={}",
                refundId, record.getRetryCount());
        } else {
            // 超过最大重试次数，转人工介入
            log.error("退款重试超限，转人工, refundId={}, retryCount={}",
                refundId, record.getRetryCount());
            recordLog(afterSaleId, "SYSTEM", "MANUAL_INTERVENTION",
                AfterSaleStatus.REFUNDING.getCode(),
                AfterSaleStatus.REFUNDING.getCode(),
                "退款重试超过" + MAX_RETRY_COUNT + "次，转人工介入");
        }
    }

    private void recordLog(Long afterSaleId, String operator, String action,
                           Integer fromStatus, Integer toStatus, String remark) {
        AfterSaleLog log = new AfterSaleLog();
        log.setAfterSaleId(afterSaleId);
        log.setOperator(operator);
        log.setAction(action);
        log.setFromStatus(fromStatus);
        log.setToStatus(toStatus);
        log.setRemark(remark);
        log.setCreateTime(new Date());
        logMapper.insert(log);
    }
```

#### 6.1.8 退款失败的重试机制

```java
/**
 * 退款重试服务
 *
 * 退款失败后自动重试，策略：
 * 1. 最大重试3次
 * 2. 退避策略：30s → 2min → 10min
 * 3. 超过最大重试次数转人工处理
 */
@Service
@Slf4j
public class RefundRetryService {

    @Autowired
    private RefundRecordMapper refundRecordMapper;

    @Autowired
    private RefundService refundService;

    @Autowired
    private AfterSaleOrderMapper afterSaleMapper;

    /** 退避间隔（秒） */
    private static final int[] RETRY_INTERVALS = {30, 120, 600};

    /** 最大重试次数 */
    private static final int MAX_RETRY = 3;

    /**
     * 定时扫描失败的退款记录，执行重试
     */
    @Scheduled(fixedRate = 30000) // 每30秒扫描一次
    public void scanAndRetry() {
        List<RefundRecord> failedRecords = refundRecordMapper
            .selectFailedForRetry(MAX_RETRY);

        for (RefundRecord record : failedRecords) {
            try {
                // 判断是否到达重试时间
                long nextRetryTime = calculateNextRetryTime(
                    record.getCreateTime(), record.getRetryCount());
                if (System.currentTimeMillis() < nextRetryTime) {
                    continue; // 未到重试时间
                }

                log.info("执行退款重试, refundId={}, retryCount={}",
                    record.getRefundId(), record.getRetryCount());

                // 重置退款记录状态为处理中
                refundRecordMapper.updateStatus(record.getRefundId(), 0);
                refundRecordMapper.incrementRetryCount(record.getRefundId());

                // 重新发起退款
                refundService.executeRefund(record.getAfterSaleId());

            } catch (Exception e) {
                log.error("退款重试异常, refundId={}", record.getRefundId(), e);
                refundRecordMapper.updateStatus(record.getRefundId(), 2); // 标记失败
                refundRecordMapper.incrementRetryCount(record.getRefundId());
            }
        }
    }

    /**
     * 计算下次重试时间（退避策略）
     */
    private long calculateNextRetryTime(Date createTime, int retryCount) {
        int intervalIndex = Math.min(retryCount, RETRY_INTERVALS.length - 1);
        int intervalSeconds = RETRY_INTERVALS[intervalIndex];
        return createTime.getTime() + (long) intervalSeconds * 1000;
    }
}
```

```java
/**
```

```java
@Service
@Slf4j
public class ReturnLogisticsService {

    @Autowired
    private AfterSaleOrderMapper afterSaleMapper;

    @Autowired
    private AfterSaleLogMapper logMapper;

    @Autowired
    private AfterSaleTimeoutService timeoutService;

    @Autowired

    private static final String LOGISTICS_LOCK_PREFIX = "lock:logistics:";

    /**
     * 用户填写退货物流单号
     */
    @Transactional(rollbackFor = Exception.class)
    public void fillLogistics(Long afterSaleId, Long userId, String logisticsNo,
                              String logisticsCompany) {
        // 幂等锁
        String lockKey = LOGISTICS_LOCK_PREFIX + afterSaleId;
        RLock lock = redissonClient.getLock(lockKey);
        try {
            if (!lock.tryLock(3, TimeUnit.SECONDS)) {
                throw new BizException("PROCESSING", "正在处理中，请稍后重试");
            }

            AfterSaleOrder afterSale = afterSaleMapper.selectByAfterSaleId(afterSaleId);
            if (afterSale == null) {
                throw new BizException("NOT_FOUND", "售后单不存在");
            }
            if (!afterSale.getUserId().equals(userId)) {
                throw new BizException("PERMISSION_DENIED", "无权操作此售后单");
            }

            // 状态校验：只有商家同意退货后才能填写物流
            if (afterSale.getStatus() != AfterSaleStatus.MERCHANT_AGREED.getCode()) {
                throw new BizException("STATUS_INVALID",
                    "当前状态不能填写物流单号");
            }

            // 校验物流单号格式
            if (StringUtils.isBlank(logisticsNo)
                || logisticsNo.length() > 64) {
                throw new BizException("LOGISTICS_NO_INVALID", "物流单号不合法");
            }

            // 幂等校验：不能重复填写
            if (StringUtils.isNotBlank(afterSale.getLogisticsNo())) {
                throw new BizException("LOGISTICS_FILLED", "已填写物流单号，不能重复填写");
            }

            // 更新物流信息
            afterSaleMapper.updateLogistics(afterSaleId, logisticsNo,
                logisticsCompany, new Date());

            // 状态变更为已退货
            afterSaleMapper.updateStatus(afterSaleId,
                AfterSaleStatus.GOODS_RETURNED.getCode());
            afterSaleMapper.updateReturnTime(afterSaleId, new Date());

            recordLog(afterSaleId, "USER", "FILL_LOGISTICS",
                AfterSaleStatus.MERCHANT_AGREED.getCode(),
                AfterSaleStatus.GOODS_RETURNED.getCode(),
                "物流公司：" + logisticsCompany + "，单号：" + logisticsNo);

            // 取消退货超时任务

            // 注册商家验货超时任务（商家7天内需确认收货）
            timeoutService.registerVerifyTimeout(afterSaleId);

            log.info("用户填写退货物流, afterSaleId={}, logisticsNo={}",
                afterSaleId, logisticsNo);

        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new BizException("SYSTEM_ERROR", "系统繁忙");
        } finally {
            if (lock.isHeldByCurrentThread()) {
                lock.unlock();
            }
        }
    }

    private void recordLog(Long afterSaleId, String operator, String action,

        logEntry.setOperator(operator);
        logEntry.setAction(action);
        logEntry.setFromStatus(fromStatus);
        logEntry.setToStatus(toStatus);
        logEntry.setRemark(remark);
        logEntry.setCreateTime(new Date());
        logMapper.insert(logEntry);
```

#### 6.2.6 各环节超时自动处理

```java
/**
 * 退货退款全链路超时处理
 *
 * 超时规则：
 * 1. 商家24h不审核 → 自动同意退货
 * 2. 用户7天不填物流 → 自动关闭
 * 3. 商家7天不确认收货 → 自动验货通过，发起退款
 * 4. 退款失败 → 自动重试3次
 */
@Service
@Slf4j
public class ReturnRefundTimeoutService {

    @Autowired
    private AfterSaleOrderMapper afterSaleMapper;

    @Autowired
    private AfterSaleLogMapper logMapper;

    @Autowired
    private RefundService refundService;

    @Scheduled(fixedRate = 60000)
    public void scanTimeout() {
        // 1. 商家超时未审核（24小时）
        scanMerchantTimeout();

        // 2. 用户超时未填物流（7天）
        scanReturnTimeout();

        // 3. 商家超时未验货（7天）
        scanVerifyTimeout();
    }

    /**
     * 商家24h未审核 → 自动同意
     */
    private void scanMerchantTimeout() {
        Date deadline = DateUtils.addHours(new Date(), -24);
        List<AfterSaleOrder> timeoutList = afterSaleMapper
            .selectByStatusAndTimeBefore(
                AfterSaleStatus.APPLIED.getCode(), deadline);

        for (AfterSaleOrder as : timeoutList) {
            try {
                autoAgree(as);
            } catch (Exception e) {
                log.error("商家超时自动同意失败, id={}",
                    as.getAfterSaleId(), e);
            }
        }
    }

    /**
     * 用户7天未填物流 → 自动关闭
     */
    private void scanReturnTimeout() {
        Date deadline = DateUtils.addDays(new Date(), -7);
        List<AfterSaleOrder> timeoutList = afterSaleMapper
            .selectByStatusAndTimeBefore(
                AfterSaleStatus.MERCHANT_AGREED.getCode(), deadline);

        for (AfterSaleOrder as : timeoutList) {
            try {
                autoClose(as);
            } catch (Exception e) {
                log.error("用户超时自动关闭失败, id={}",
                    as.getAfterSaleId(), e);
            }
        }
    }

    /**
     * 商家7天未验货 → 自动验货通过，发起退款
     */
    private void scanVerifyTimeout() {
        Date deadline = DateUtils.addDays(new Date(), -7);
        List<AfterSaleOrder> timeoutList = afterSaleMapper
            .selectByStatusAndTimeBefore(
                AfterSaleStatus.GOODS_RETURNED.getCode(), deadline);

        for (AfterSaleOrder as : timeoutList) {
            try {
                autoVerifyAndRefund(as);
            } catch (Exception e) {
                log.error("商家超时自动验货失败, id={}",
                    as.getAfterSaleId(), e);
            }
        }
    }

    @Transactional(rollbackFor = Exception.class)
    private void autoAgree(AfterSaleOrder as) {
        // 二次校验状态（防止并发导致状态已变更）
        AfterSaleOrder latest = afterSaleMapper.selectByAfterSaleId(
            as.getAfterSaleId());
        if (latest.getStatus() != AfterSaleStatus.APPLIED.getCode()) {
            return;
        }

        afterSaleMapper.updateStatus(as.getAfterSaleId(),
            AfterSaleStatus.MERCHANT_AGREED.getCode());
        afterSaleMapper.updateAgreeTime(as.getAfterSaleId(), new Date());

        recordLog(as.getAfterSaleId(), "SYSTEM", "AUTO_AGREE_TIMEOUT",
            AfterSaleStatus.APPLIED.getCode(),
            AfterSaleStatus.MERCHANT_AGREED.getCode(),
            "商家24小时未审核，系统自动同意");

        // 退货退款：注册退货超时任务
        // （仅退款：直接发起退款）
        if (as.getType() == AfterSaleType.REFUND_ONLY.getCode()) {
            refundService.executeRefund(as.getAfterSaleId());
        }
    }

    @Transactional(rollbackFor = Exception.class)
    private void autoClose(AfterSaleOrder as) {
        AfterSaleOrder latest = afterSaleMapper.selectByAfterSaleId(
            as.getAfterSaleId());
        if (latest.getStatus() != AfterSaleStatus.MERCHANT_AGREED.getCode()) {
            return;
        }

        afterSaleMapper.updateStatus(as.getAfterSaleId(),
            AfterSaleStatus.CLOSED.getCode());

        recordLog(as.getAfterSaleId(), "SYSTEM", "AUTO_CLOSE_TIMEOUT",
            AfterSaleStatus.MERCHANT_AGREED.getCode(),
            AfterSaleStatus.CLOSED.getCode(),
            "用户7天未填写退货物流，系统自动关闭");
    }

    @Transactional(rollbackFor = Exception.class)
    private void autoVerifyAndRefund(AfterSaleOrder as) {
        AfterSaleOrder latest = afterSaleMapper.selectByAfterSaleId(
            as.getAfterSaleId());
        if (latest.getStatus() != AfterSaleStatus.GOODS_RETURNED.getCode()) {
            return;
        }

        // 自动验货通过
        afterSaleMapper.updateStatus(as.getAfterSaleId(),
            AfterSaleStatus.VERIFIED.getCode());
        afterSaleMapper.updateVerifyTime(as.getAfterSaleId(), new Date());

        recordLog(as.getAfterSaleId(), "SYSTEM", "AUTO_VERIFY_TIMEOUT",
            AfterSaleStatus.GOODS_RETURNED.getCode(),
            AfterSaleStatus.VERIFIED.getCode(),
            "商家7天未确认收货，系统自动验货通过");

        // 发起退款
        refundService.executeRefund(as.getAfterSaleId());
    }

    private void recordLog(Long afterSaleId, String operator, String action,
                           Integer fromStatus, Integer toStatus, String remark) {
        AfterSaleLog logEntry = new AfterSaleLog();
        logEntry.setAfterSaleId(afterSaleId);
        logEntry.setOperator(operator);
        logEntry.setAction(action);
        logEntry.setFromStatus(fromStatus);
        logEntry.setToStatus(toStatus);
        logEntry.setRemark(remark);
        logEntry.setCreateTime(new Date());
        logMapper.insert(logEntry);
    }
}
```

### 案例三：退款对账与异常处理全链路

#### 6.3.1 场景描述

每日凌晨执行退款对账任务，将平台退款记录与支付渠道退款记录逐笔核对，发现差异并自动处理。同时扫描长期未完结的售后单，触发人工介入。

```
每日对账定时任务 → 拉取平台退款记录 → 拉取支付渠道退款记录 → 逐笔比对
→ 差异分类处理（平台有/渠道无、平台无/渠道有、金额不一致）
→ 长期未完结售后单扫描 → 人工介入通知
```

#### 6.3.2 每日退款对账任务

```java
/**
 * 退款对账服务
 *
 * 每日凌晨2点执行，核对前一天的退款记录
 * 对账维度：按退款流水号(refund_id)逐笔比对
 */
@Service
@Slf4j
public class RefundReconcileService {

    @Autowired
    private RefundRecordMapper refundRecordMapper;

    @Autowired
    private PaymentFeignClient paymentClient;

    @Autowired
    private AfterSaleOrderMapper afterSaleMapper;

    @Autowired
    private ReconcileDiffMapper reconcileDiffMapper;

    @Autowired
    private AfterSaleLogMapper logMapper;

    @Autowired
    private NotificationFeignClient notificationClient;

    /**
     * 每日凌晨2点执行对账
     */
    @Scheduled(cron = "0 0 2 * * ?")
    public void dailyReconcile() {
        Date yesterday = DateUtils.addDays(new Date(), -1);
        Date startDate = DateUtils.getStartOfDay(yesterday);
        Date endDate = DateUtils.getEndOfDay(yesterday);

        log.info("开始退款对账, period=[{} ~ {}]", startDate, endDate);

        try {
            // Step 1: 拉取平台退款记录
            List<RefundRecord> platformRecords = refundRecordMapper
                .selectByTimeRange(startDate, endDate);
            log.info("平台退款记录数: {}", platformRecords.size());

            // Step 2: 拉取支付渠道退款记录
            List<ChannelRefundRecord> channelRecords = paymentClient
                .queryRefundByDate(startDate, endDate);
            log.info("支付渠道退款记录数: {}", channelRecords.size());

            // Step 3: 逐笔比对
            ReconcileResult result = reconcile(platformRecords, channelRecords);

            // Step 4: 处理差异
            handleDifferences(result);

            // Step 5: 生成对账报告
            generateReport(startDate, endDate, result);

            log.info("退款对账完成, 总数={}, 一致={}, 差异={}",
                result.getTotal(), result.getMatched(), result.getDiffCount());

        } catch (Exception e) {
            log.error("退款对账异常", e);
            // 发送告警通知
            notificationClient.sendAlert("退款对账异常", e.getMessage());
        }
    }

    /**
     * 逐笔比对
     */
    private ReconcileResult reconcile(List<RefundRecord> platformRecords,
                                       List<ChannelRefundRecord> channelRecords) {
        ReconcileResult result = new ReconcileResult();
        result.setTotal(platformRecords.size());

        // 以退款流水号为key建立map
        Map<String, ChannelRefundRecord> channelMap = channelRecords.stream()
            .collect(Collectors.toMap(
                ChannelRefundRecord::getRefundNo, r -> r, (a, b) -> b));

        for (RefundRecord platform : platformRecords) {
            String refundNo = String.valueOf(platform.getRefundId());
            ChannelRefundRecord channel = channelMap.get(refundNo);

            if (channel == null) {
                // 差异类型1：平台有退款记录，支付渠道无
                result.addDiff(buildDiff(platform, null,
                    DiffType.PLATFORM_ONLY));
            } else if (platform.getStatus() != channel.getStatus()) {
                // 差异类型2：状态不一致
                result.addDiff(buildDiff(platform, channel,
                    DiffType.STATUS_MISMATCH));
            } else if (platform.getRefundAmount() != channel.getRefundAmount()) {
                // 差异类型3：金额不一致
                result.addDiff(buildDiff(platform, channel,
                    DiffType.AMOUNT_MISMATCH));
            } else {
                result.incrementMatched();
            }
        }

        // 检查渠道有但平台无的记录
        Set<String> platformRefundNos = platformRecords.stream()
            .map(r -> String.valueOf(r.getRefundId()))
            .collect(Collectors.toSet());

        for (ChannelRefundRecord channel : channelRecords) {
            if (!platformRefundNos.contains(channel.getRefundNo())) {
                result.addDiff(buildDiff(null, channel,
                    DiffType.CHANNEL_ONLY));
            }
        }

        return result;
    }

    /**
     * 处理差异
     */
    private void handleDifferences(ReconcileResult result) {
        for (ReconcileDiff diff : result.getDiffs()) {
            try {
                handleSingleDiff(diff);
            } catch (Exception e) {
                log.error("处理对账差异异常, diffId={}", diff.getId(), e);
            }
        }
    }

    /**
     * 处理单条差异
     */
    @Transactional(rollbackFor = Exception.class)
    private void handleSingleDiff(ReconcileDiff diff) {
        switch (diff.getDiffType()) {
            case PLATFORM_ONLY:
                // 平台有退款记录，渠道无
                // 可能：退款请求未到达渠道 / 渠道处理超时
                handlePlatformOnly(diff);
                break;

            case CHANNEL_ONLY:
                // 渠道有退款记录，平台无
                // 可能：渠道手动退款 / 平台数据丢失
                handleChannelOnly(diff);
                break;

            case STATUS_MISMATCH:
                // 状态不一致（如平台显示失败但渠道已退款成功）
                handleStatusMismatch(diff);
                break;

            case AMOUNT_MISMATCH:
                // 金额不一致
                handleAmountMismatch(diff);
                break;

            default:
                log.warn("未知差异类型, diffType={}", diff.getDiffType());
        }

        // 保存差异记录
        reconcileDiffMapper.insert(diff);
    }

    /**
     * 差异处理：平台有退款记录，渠道无
     *
     * 场景：退款请求提交到支付渠道但渠道未处理，或网络超时
     * 处理：重新查询渠道退款状态，若确实未退款则重新发起
     */
    private void handlePlatformOnly(ReconcileDiff diff) {
        Long refundId = diff.getPlatformRefundId();

        // 重新查询支付渠道
        PaymentRefundQueryResult queryResult = paymentClient.queryRefund(refundId);

        if (queryResult == null || !queryResult.isExists()) {
            // 渠道确实没有退款记录，需要重新发起
            log.warn("渠道无退款记录，需重新发起, refundId={}", refundId);

            RefundRecord record = refundRecordMapper.selectByRefundId(refundId);
            if (record != null && record.getStatus() != 1) {
                // 重置状态为处理中，等待重试任务重新发起
                refundRecordMapper.updateStatus(refundId, 0);
                refundRecordMapper.updateRetryCount(refundId, 0);

                recordLog(record.getAfterSaleId(), "SYSTEM", "RECONCILE_RETRY",
                    "对账发现渠道无退款记录，重新发起退款");
            }
        } else {
            // 渠道有记录但对账文件中没有，补录即可
            log.info("渠道有退款记录但未在对账文件中, refundId={}", refundId);
            refundRecordMapper.updateTradeNo(refundId, queryResult.getTradeNo());
        }
    }

    /**
     * 差异处理：渠道有退款记录，平台无
     *
     * 场景：渠道侧手动发起退款（如客服通过渠道后台操作）
     * 处理：补录平台退款记录，更新售后单状态
     */
    private void handleChannelOnly(ReconcileDiff diff) {
        String refundNo = diff.getChannelRefundNo();
        log.warn("渠道有退款但平台无记录, refundNo={}", refundNo);

        // 尝试通过订单号关联
        ChannelRefundRecord channelRecord = paymentClient
            .queryRefundByNo(refundNo);

        if (channelRecord != null) {
            // 查询关联的售后单
            AfterSaleOrder afterSale = afterSaleMapper
                .selectByOrderId(channelRecord.getOrderId());

            if (afterSale != null
                && afterSale.getStatus() == AfterSaleStatus.REFUNDING.getCode()) {
                // 补录退款记录
                RefundRecord record = new RefundRecord();
                record.setRefundId(Long.valueOf(refundNo));
                record.setAfterSaleId(afterSale.getAfterSaleId());
                record.setOrderId(afterSale.getOrderId());
                record.setUserId(afterSale.getUserId());
                record.setRefundAmount(channelRecord.getRefundAmount());
                record.setRefundType("ORIGINAL");
                record.setStatus(channelRecord.isSuccess() ? 1 : 2);
                record.setTradeNo(channelRecord.getTradeNo());
                record.setRefundNo(refundNo);
                record.setCreateTime(new Date());
                refundRecordMapper.insert(record);

                if (channelRecord.isSuccess()) {
                    afterSaleMapper.updateStatus(afterSale.getAfterSaleId(),
                        AfterSaleStatus.COMPLETED.getCode());
                    afterSaleMapper.updateCompleteTime(
                        afterSale.getAfterSaleId(), new Date());

                    recordLog(afterSale.getAfterSaleId(), "SYSTEM",
                        "RECONCILE_COMPLETE",
                        "对账补录：渠道已退款成功，更新售后单为已完成");
                }
            } else {
                // 无法关联售后单，发人工处理
                notificationClient.sendAlert("对账异常-无法关联售后单",
                    "退款单号：" + refundNo);
            }
        }
    }

    /**
     * 差异处理：状态不一致
     *
     * 场景：平台显示退款失败但渠道已退款成功（回调丢失）
     * 处理：以渠道状态为准，更新平台状态
     */
    private void handleStatusMismatch(ReconcileDiff diff) {
        Long refundId = diff.getPlatformRefundId();
        log.warn("退款状态不一致, refundId={}, platformStatus={}, channelStatus={}",
            refundId, diff.getPlatformStatus(), diff.getChannelStatus());

        RefundRecord record = refundRecordMapper.selectByRefundId(refundId);

        if (diff.getChannelStatus() == 1 && record.getStatus() != 1) {
            // 渠道成功，平台未更新 → 回调丢失，补处理
            log.info("回调丢失，补处理退款成功, refundId={}", refundId);

            refundRecordMapper.updateStatus(refundId, 1);
            refundRecordMapper.updateCompleteTime(refundId, new Date());

            // 更新售后单状态
            afterSaleMapper.updateStatus(record.getAfterSaleId(),
                AfterSaleStatus.COMPLETED.getCode());
            afterSaleMapper.updateCompleteTime(
                record.getAfterSaleId(), new Date());

            recordLog(record.getAfterSaleId(), "SYSTEM", "RECONCILE_FIX_STATUS",
                "对账修复：渠道已退款成功（回调丢失），更新平台状态");

        } else if (diff.getChannelStatus() == 2 && record.getStatus() == 1) {
            // 渠道失败，平台显示成功 → 严重异常，需人工处理
            log.error("严重异常：渠道退款失败但平台显示成功, refundId={}", refundId);
            notificationClient.sendAlert("对账严重异常",
                "退款单号：" + refundId + "，渠道退款失败但平台显示成功，需人工核实");
        }
    }

    /**
     * 差异处理：金额不一致
     *
     * 场景：平台退款金额与渠道退款金额不一致
     * 处理：全部转人工处理
     */
    private void handleAmountMismatch(ReconcileDiff diff) {
        log.error("退款金额不一致, refundId={}, platform={}, channel={}",
            diff.getPlatformRefundId(),
            diff.getPlatformAmount(), diff.getChannelAmount());

        notificationClient.sendAlert("对账异常-金额不一致",
            String.format("退款单号：%d，平台金额：%d，渠道金额：%d",
                diff.getPlatformRefundId(),
                diff.getPlatformAmount(), diff.getChannelAmount()));
    }

    private void recordLog(Long afterSaleId, String operator, String action,
                           String remark) {
        AfterSaleLog logEntry = new AfterSaleLog();
        logEntry.setAfterSaleId(afterSaleId);
        logEntry.setOperator(operator);
        logEntry.setAction(action);
        logEntry.setRemark(remark);
        logEntry.setCreateTime(new Date());
        logMapper.insert(logEntry);
    }

    private ReconcileDiff buildDiff(RefundRecord platform,
                                      ChannelRefundRecord channel,
                                      DiffType diffType) {
        ReconcileDiff diff = new ReconcileDiff();
        diff.setDiffType(diffType);
        if (platform != null) {
            diff.setPlatformRefundId(platform.getRefundId());
            diff.setPlatformStatus(platform.getStatus());
            diff.setPlatformAmount(platform.getRefundAmount());
        }
        if (channel != null) {
            diff.setChannelRefundNo(channel.getRefundNo());
            diff.setChannelStatus(channel.isSuccess() ? 1 : 2);
            diff.setChannelAmount(channel.getRefundAmount());
        }
        diff.setCreateTime(new Date());
        return diff;
    }
}
```

#### 6.3.3 长期未完结售后单处理

```java
/**
 * 长期未完结售后单扫描服务
 *
 * 每天扫描超过3天仍未完结的售后单，自动处理或转人工
 */
@Service
@Slf4j
public class StaleAfterSaleService {

    @Autowired
    private AfterSaleOrderMapper afterSaleMapper;

    @Autowired
    private AfterSaleLogMapper logMapper;

    @Autowired
    private RefundRecordMapper refundRecordMapper;

    @Autowired
    private NotificationFeignClient notificationClient;

    @Autowired
    private RefundService refundService;

    /** 超期天数阈值 */
    private static final int STALE_DAYS = 3;

    /**
     * 每天上午10点扫描长期未完结的售后单
     */
    @Scheduled(cron = "0 0 10 * * ?")
    public void scanStaleOrders() {
        Date deadline = DateUtils.addDays(new Date(), -STALE_DAYS);
        List<AfterSaleOrder> staleList = afterSaleMapper
            .selectStaleOrders(deadline);

        log.info("扫描到长期未完结售后单: {} 个", staleList.size());

        for (AfterSaleOrder as : staleList) {
            try {
                handleStaleOrder(as);
            } catch (Exception e) {
                log.error("处理长期未完结售后单异常, afterSaleId={}",
                    as.getAfterSaleId(), e);
            }
        }
    }

    /**
     * 处理单个长期未完结售后单
     */
    @Transactional(rollbackFor = Exception.class)
    private void handleStaleOrder(AfterSaleOrder as) {
        log.info("处理长期未完结售后单, afterSaleId={}, status={}, applyTime={}",
            as.getAfterSaleId(), as.getStatus(), as.getApplyTime());

        switch (AfterSaleStatus.fromCode(as.getStatus())) {
            case APPLIED:
                // 长期待审核：商家可能遗漏，发催办通知
                notificationClient.sendMerchantNotify(as.getAfterSaleId(),
                    "您有售后单已超过" + STALE_DAYS + "天未审核，请及时处理");
                recordLog(as.getAfterSaleId(), "SYSTEM", "STALE_REMIND",
                    "售后单长期未审核，已发送催办通知");
                break;

            case MERCHANT_AGREED:
                // 商家已同意但用户未退货：检查是否超时
                if (as.getType() == AfterSaleType.RETURN_REFUND.getCode()) {
                    notificationClient.sendUserNotify(as.getAfterSaleId(),
                        as.getUserId(),
                        "您的退货申请已通过，请尽快寄回商品");
                    recordLog(as.getAfterSaleId(), "SYSTEM", "STALE_REMIND",
                        "用户长期未退货，已发送提醒通知");
                }
                break;

            case GOODS_RETURNED:
                // 已退货但商家未验货：发催办通知
                notificationClient.sendMerchantNotify(as.getAfterSaleId(),
                    "用户已退货，请尽快确认收货");
                recordLog(as.getAfterSaleId(), "SYSTEM", "STALE_REMIND",
                    "商家长期未验货，已发送催办通知");
                break;

            case REFUNDING:
                // 退款中：检查退款记录状态
                handleStuckRefund(as);
                break;

            default:
                log.debug("无需处理的售后状态, afterSaleId={}, status={}",
                    as.getAfterSaleId(), as.getStatus());
        }
    }

    /**
     * 处理卡在退款中的售后单
     */
    private void handleStuckRefund(AfterSaleOrder as) {
        RefundRecord record = refundRecordMapper
            .selectByAfterSaleId(as.getAfterSaleId());

        if (record == null) {
            // 退款中但没有退款记录，异常情况
            log.error("退款中但无退款记录, afterSaleId={}", as.getAfterSaleId());
            notificationClient.sendAlert("售后异常",
                "售后单" + as.getAfterSaleId() + "处于退款中但无退款记录");
            return;
        }

        if (record.getStatus() == 0) {
            // 退款记录状态为处理中，但已超期
            // 主动查询支付渠道退款状态
            PaymentRefundQueryResult queryResult = paymentClient
                .queryRefund(record.getRefundId());

            if (queryResult != null && queryResult.isSuccess()) {
                // 渠道已退款成功，回调丢失，补处理
                log.info("发现回调丢失，补处理, afterSaleId={}, refundId={}",
                    as.getAfterSaleId(), record.getRefundId());

                refundRecordMapper.updateStatus(record.getRefundId(), 1);
                refundRecordMapper.updateCompleteTime(
                    record.getRefundId(), new Date());
                afterSaleMapper.updateStatus(as.getAfterSaleId(),
                    AfterSaleStatus.COMPLETED.getCode());
                afterSaleMapper.updateCompleteTime(
                    as.getAfterSaleId(), new Date());

                recordLog(as.getAfterSaleId(), "SYSTEM", "STALE_FIX",
                    "长期未完结扫描发现回调丢失，补处理退款成功");

            } else if (record.getRetryCount() >= 3) {
                // 超过最大重试次数，转人工
                log.error("退款重试超限，转人工, afterSaleId={}",
                    as.getAfterSaleId());
                notificationClient.sendAlert("退款超限转人工",
                    "售后单" + as.getAfterSaleId()
                        + "退款重试超过3次，需人工介入");
                recordLog(as.getAfterSaleId(), "SYSTEM", "MANUAL_INTERVENTION",
                    "退款重试超限，已转人工介入");
            } else {
                // 未超重试次数，触发重试
                log.info("触发退款重试, afterSaleId={}", as.getAfterSaleId());
                refundService.executeRefund(as.getAfterSaleId());
            }
        }
    }

    private void recordLog(Long afterSaleId, String operator, String action,
                           String remark) {
        AfterSaleLog logEntry = new AfterSaleLog();
        logEntry.setAfterSaleId(afterSaleId);
        logEntry.setOperator(operator);
        logEntry.setAction(action);
        logEntry.setRemark(remark);
        logEntry.setCreateTime(new Date());
        logMapper.insert(logEntry);
    }
}
```

#### 6.3.4 异常退款人工介入机制

```java
/**
 * 人工介入管理服务
 *
 * 当系统自动处理无法解决的异常发生时，创建人工介入工单，
 * 通知客服/运营人员处理
 */
@Service
@Slf4j
public class ManualInterventionService {

    @Autowired
    private ManualTaskMapper manualTaskMapper;

    @Autowired
    private AfterSaleOrderMapper afterSaleMapper;

    @Autowired
    private RefundRecordMapper refundRecordMapper;

    @Autowired
    private NotificationFeignClient notificationClient;

    @Autowired
    private IdGenerator idGenerator;

    /**
     * 创建人工介入工单
     */
    @Transactional(rollbackFor = Exception.class)
    public Long createManualTask(Long afterSaleId, ManualTaskType type,
                                  String description, int priority) {
        // 幂等校验：同一售后单同一类型不重复创建
        ManualTask existTask = manualTaskMapper
            .selectActiveByAfterSaleAndType(afterSaleId, type);
        if (existTask != null) {
            log.info("人工介入工单已存在, afterSaleId={}, taskId={}",
                afterSaleId, existTask.getTaskId());
            return existTask.getTaskId();
        }

        Long taskId = idGenerator.nextId();
        ManualTask task = new ManualTask();
        task.setTaskId(taskId);
        task.setAfterSaleId(afterSaleId);
        task.setType(type.getCode());
        task.setDescription(description);
        task.setPriority(priority);
        task.setStatus(0); // 待处理
        task.setCreateTime(new Date());
        manualTaskMapper.insert(task);

        // 发送通知给客服团队
        notificationClient.sendManualTaskNotify(taskId, afterSaleId,
            type.getDescription(), description, priority);

        log.info("创建人工介入工单, taskId={}, afterSaleId={}, type={}",
            taskId, afterSaleId, type);

        return taskId;
    }

    /**
     * 客服处理人工介入工单
     */
    @Transactional(rollbackFor = Exception.class)
    public void handleManualTask(Long taskId, Long handlerId,
                                  ManualHandleResult result) {
        ManualTask task = manualTaskMapper.selectByTaskId(taskId);
        if (task == null) {
            throw new BizException("TASK_NOT_FOUND", "工单不存在");
        }
        if (task.getStatus() != 0) {
            throw new BizException("TASK_HANDLED", "工单已处理");
        }

        // 更新工单状态
        task.setHandlerId(handlerId);
        task.setHandleResult(result.getAction());
        task.setHandleRemark(result.getRemark());
        task.setStatus(1); // 已处理
        task.setHandleTime(new Date());
        manualTaskMapper.updateById(task);

        // 根据处理结果执行对应操作
        switch (result.getAction()) {
            case "FORCE_REFUND":
                // 强制退款
                forceRefund(task.getAfterSaleId());
                break;

            case "CLOSE_AFTER_SALE":
                // 关闭售后单
                forceClose(task.getAfterSaleId(), result.getRemark());
                break;

            case "RETRY_REFUND":
                // 重新退款
                refundService.executeRefund(task.getAfterSaleId());
                break;

            case "ADJUST_AMOUNT":
                // 调整退款金额后重新退款
                adjustAmountAndRefund(task.getAfterSaleId(),
                    result.getNewAmount());
                break;

            default:
                log.info("人工处理操作: {}", result.getAction());
        }

        log.info("人工介入工单处理完成, taskId={}, handlerId={}, action={}",
            taskId, handlerId, result.getAction());
    }

    /**
     * 强制退款（跳过状态校验）
     */
    private void forceRefund(Long afterSaleId) {
        AfterSaleOrder afterSale = afterSaleMapper.selectByAfterSaleId(afterSaleId);
        if (afterSale == null) {
            throw new BizException("NOT_FOUND", "售后单不存在");
        }

        // 检查是否已退款成功
        RefundRecord record = refundRecordMapper.selectByAfterSaleId(afterSaleId);
        if (record != null && record.getStatus() == 1) {
            log.info("已退款成功，无需强制退款, afterSaleId={}", afterSaleId);
            return;
        }

        // 直接发起退款
        refundService.executeRefund(afterSaleId);
    }

    /**
     * 强制关闭售后单
     */
    private void forceClose(Long afterSaleId, String reason) {
        AfterSaleOrder afterSale = afterSaleMapper.selectByAfterSaleId(afterSaleId);
        if (afterSale == null) {
            throw new BizException("NOT_FOUND", "售后单不存在");
        }

        afterSaleMapper.updateStatus(afterSaleId,
            AfterSaleStatus.CLOSED.getCode());

        AfterSaleLog logEntry = new AfterSaleLog();
        logEntry.setAfterSaleId(afterSaleId);
        logEntry.setOperator("MANUAL");
        logEntry.setAction("FORCE_CLOSE");
        logEntry.setFromStatus(afterSale.getStatus());
        logEntry.setToStatus(AfterSaleStatus.CLOSED.getCode());
        logEntry.setRemark("人工强制关闭，原因：" + reason);
        logEntry.setCreateTime(new Date());
        logMapper.insert(logEntry);

        log.info("人工强制关闭售后单, afterSaleId={}, reason={}",
            afterSaleId, reason);
    }

    /**
     * 调整退款金额并重新退款
     */
    private void adjustAmountAndRefund(Long afterSaleId, int newAmount) {
        AfterSaleOrder afterSale = afterSaleMapper.selectByAfterSaleId(afterSaleId);
        if (afterSale == null) {
            throw new BizException("NOT_FOUND", "售后单不存在");
        }

        // 校验新金额
        OrderVO order = orderClient.getOrder(afterSale.getOrderId());
        if (newAmount > order.getPaidAmount()) {
            throw new BizException("AMOUNT_EXCEED", "退款金额不能超过实付金额");
        }
        if (newAmount <= 0) {
            throw new BizException("AMOUNT_INVALID", "退款金额必须大于0");
        }

        // 更新退款金额
        afterSaleMapper.updateRefundAmount(afterSaleId, newAmount);

        AfterSaleLog logEntry = new AfterSaleLog();
        logEntry.setAfterSaleId(afterSaleId);
        logEntry.setOperator("MANUAL");
        logEntry.setAction("ADJUST_AMOUNT");
        logEntry.setRemark("人工调整退款金额，原金额：" + afterSale.getRefundAmount()
            + "，新金额：" + newAmount);
        logEntry.setCreateTime(new Date());
        logMapper.insert(logEntry);

        // 重新发起退款
        refundService.executeRefund(afterSaleId);
    }
}

/**
 * 人工介入工单类型
 */
public enum ManualTaskType {
    REFUND_RETRY_EXCEED("退款重试超限", 1),
    RECONCILE_DIFF("对账差异", 2),
    AMOUNT_MISMATCH("退款金额不一致", 1),
    STATUS_ANOMALY("状态异常", 1),
    LONG_STALE("长期未完结", 3),
    CHANNEL_MANUAL_REFUND("渠道手动退款", 2);

    private final String description;
    private final int defaultPriority;

    ManualTaskType(String description, int defaultPriority) {
        this.description = description;
        this.defaultPriority = defaultPriority;
    }

    public String getDescription() { return description; }
    public int getDefaultPriority() { return defaultPriority; }
    public int getCode() { return ordinal() + 1; }
}
```
